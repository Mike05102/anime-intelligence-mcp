function decideGenericPurchase(domain,evaluated,intent){
  const sorted=[...evaluated].sort((a,b)=>Number(b?.purchase_evaluation?.score||-999)-Number(a?.purchase_evaluation?.score||-999));
  const best=sorted.find(x=>x?.purchase_evaluation?.decision_ready)||sorted[0]||null;
  if(!best)return {action:"NO_EVIDENCE",can_purchase:false,reason_codes:["no_offer_evidence"],best:null};
  const e=best.purchase_evaluation||{},u=e.unknowns||[];
  if(e.hard_blockers?.length)return {action:"VERIFY_BEFORE_BUY",can_purchase:false,reason_codes:e.hard_blockers,best};
  if(e.identity?.known===false)return {action:"COMPARE_AND_CONFIRM_IDENTITY",can_purchase:false,reason_codes:["exact_identity_not_proven"],best};
  if(u.some(x=>["shipping","delivery_time","stock_status"].includes(x)))return {action:"VERIFY_FULFILLMENT",can_purchase:false,reason_codes:u.filter(x=>["shipping","delivery_time","stock_status"].includes(x)),best};
  if(u.some(x=>["tax_or_vat","customs_or_duty","brokerage_if_applicable","tax_or_vat_or_duty_if_cross_border"].includes(x)))return {action:"VERIFY_TOTAL_COST",can_purchase:false,reason_codes:u.filter(x=>["tax_or_vat","customs_or_duty","brokerage_if_applicable","tax_or_vat_or_duty_if_cross_border"].includes(x)),best};
  return {action:"BUY_RECOMMENDED",can_purchase:true,reason_codes:["identity_and_hard_constraints_passed","fulfillment_and_known_total_sufficient","evidence_ranked_offer"],best};
}

function universalEquivalentMatch(phrase,title){
  const a=normalize(phrase),b=normalize(title);if(!a||!b)return false;
  const group=UNIVERSAL_QUERY_EQUIVALENTS.find(g=>g.some(x=>normalize(x)===a));
  return (group||[phrase]).some(x=>b.includes(normalize(x)));
}
function universalCandidateRelevance(query="",o={}){
  const q=normalize(query),title=String(o?.title||"");if(!q||!title)return 0;
  const direct=labelSimilarityScore(q,title);
  const words=String(query).trim().split(/\s+/).filter(Boolean);
  // Only award synonym relevance for real query concepts present in the listing.
  const conceptHits=words.filter(w=>w.length>=3&&universalEquivalentMatch(w,title)).length;
  const phrases=UNIVERSAL_QUERY_EQUIVALENTS.filter(g=>g.some(x=>q.includes(normalize(x)))&&g.some(x=>normalize(title).includes(normalize(x))));
  const matchedConcepts=new Set([...phrases.map(g=>g[0]),...words.filter(w=>w.length>=3&&universalEquivalentMatch(w,title))]);
  const brand=String(o?.brand||"");
  const brandInTitle=brand&&normalize(title).includes(normalize(brand));
  const exactIdentifier=validGtin(digitsOnly(o?.gtin||o?.jan_code||""));
  const lexical=Math.min(75,matchedConcepts.size*24+(brandInTitle?4:0));
  return Math.min(100,Math.max(direct,lexical)+(exactIdentifier?3:0));
}

  if(brandConflict&&(sim==null||sim<UNIVERSAL_DISCOVERY_POLICY.canonical_conflict_brand_floor))reasons.push("manufacturer_conflict");
  return {conflict:reasons.length>0,reasons,title_similarity:sim,existing_manufacturer:existing?.manufacturer||null,new_manufacturer:row?.manufacturer||null};

function adaptiveExplorationScore(candidate,history,now=Date.now()){
  const family=universalQueryFamily(candidate.domain,candidate.query),st=history?.families?.get(family),domain=history?.domains?.get(candidate.domain),runs=Number(st?.runs||0),uy=st?Number(st.unique_yield||0):1,gy=st?Number(st.gtin_yield||0):0,cy=st?Number(st.canonical_yield||0):0,my=st?Number(st.merchant_evidence_yield||0):0,mty=st?Number(st.merchant_trust_yield||0):0,py=st?Number(st.purchase_ready_yield||0):0,dy=st?Number(st.decision_ready_yield||0):0,ay=st?Number(st.decision_answerability_yield||0):0,ce=st?Number(st.canonical_per_provider_call||0):0,dup=st?Number(st.duplicate_ratio||0):0,age=st?Number(st.age_hours||0):Infinity;
  const sourceBase={demand:3.2,frontier:1.6,stale_refresh:1.0,exploration:1.7,long_tail:1.95,domain_gap:2.20}[candidate.source]||1.5;
  const total=Math.max(1,Number(history?.totalRuns||0)),ucb=Math.sqrt(Math.log(total+2)/(runs+1));
  const domainRuns=Number(domain?.runs||0),domainNovelty=1/Math.sqrt(domainRuns+1);
  let score=sourceBase+UNIVERSAL_DISCOVERY_POLICY.unique_yield_weight*uy+UNIVERSAL_DISCOVERY_POLICY.gtin_yield_weight*gy+UNIVERSAL_DISCOVERY_POLICY.canonical_yield_weight*cy+UNIVERSAL_DISCOVERY_POLICY.merchant_evidence_yield_weight*my+UNIVERSAL_DISCOVERY_POLICY.merchant_trust_yield_weight*mty+UNIVERSAL_DISCOVERY_POLICY.purchase_ready_yield_weight*py+UNIVERSAL_DISCOVERY_POLICY.decision_ready_yield_weight*dy+UNIVERSAL_DISCOVERY_POLICY.decision_answerability_weight*ay+UNIVERSAL_DISCOVERY_POLICY.canonical_efficiency_weight*Math.min(1,ce)+UNIVERSAL_DISCOVERY_POLICY.exploration_bonus_weight*ucb+0.45*domainNovelty-UNIVERSAL_DISCOVERY_POLICY.duplicate_penalty_weight*dup;
  if(runs>=UNIVERSAL_DISCOVERY_POLICY.max_query_family_runs_7d)score-=0.8;
  if(age<UNIVERSAL_DISCOVERY_POLICY.query_cooldown_hours)score-=1.0*(1-age/UNIVERSAL_DISCOVERY_POLICY.query_cooldown_hours);
  if(st&&uy<0.35&&age<UNIVERSAL_DISCOVERY_POLICY.low_yield_cooldown_hours)score-=1.2*(1-age/UNIVERSAL_DISCOVERY_POLICY.low_yield_cooldown_hours);
  if(st&&Number(st.candidates||0)>=50&&cy===0)score-=UNIVERSAL_DISCOVERY_POLICY.zero_canonical_penalty;if(st&&Number(st.candidates||0)>=50&&dy===0)score-=UNIVERSAL_DISCOVERY_POLICY.zero_decision_ready_penalty;
  const paidConversions=Number(history?.paidFamilies?.get(family)||0);if(paidConversions>0)score+=UNIVERSAL_DISCOVERY_POLICY.revenue_signal_weight*Math.min(2,Math.log2(1+paidConversions));
  if(candidate.pre_payment)score+=1.0;
  return {score:Math.round(score*1000)/1000,family,history_runs:runs,history_candidates:Number(st?.candidates||0),history_gtin_candidates:Number(st?.gtins||0),history_canonical_new_count:Number(st?.canonical_new||0),history_avg_candidates_per_run:runs?Math.round(Number(st?.candidates||0)/runs*1000)/1000:0,history_unique_yield:st?Math.round(uy*1000)/1000:null,history_duplicate_ratio:st?Math.round(dup*1000)/1000:null,history_gtin_yield:st?Math.round(gy*1000)/1000:null,history_canonical_yield:st?Math.round(cy*1000)/1000:null,history_merchant_evidence_yield:st?Math.round(my*1000)/1000:null,history_merchant_trust_yield:st?Math.round(mty*1000)/1000:null,history_purchase_ready_yield:st?Math.round(py*1000)/1000:null,history_decision_ready_yield:st?Math.round(dy*1000)/1000:null,history_decision_answerability_yield:st?Math.round(ay*1000)/1000:null,history_canonical_per_provider_call:st?Math.round(ce*1000)/1000:null,history_paid_conversions:paidConversions,last_run_age_hours:Number.isFinite(age)?Math.round(age*10)/10:null};
}

function universalIdentityRoleScore(x={}){
  const n=Math.max(0,Number(x.history_candidates||0)),runs=Math.max(0,Number(x.history_runs||0));
  const gtins=Math.max(0,Number(x.history_gtin_candidates||0)),canonical=Math.max(0,Number(x.history_canonical_new_count||0));
  const avg=Math.max(0,Number(x.history_avg_candidates_per_run||(runs?n/runs:0)));
  const uy=Number(x.history_unique_yield??0.5),ce=Number(x.history_canonical_per_provider_call||0);
  // Beta-style shrinkage prevents a 1/1 exact listing from outranking a broad query that yields many canonical identities.
  const gy=(gtins+1.5)/(n+12.5),cy=(canonical+0.75)/(n+12.5);
  const volume=Math.min(1,Math.log1p(avg)/Math.log(31));
  const expectedCanonical=Math.min(1,(cy*Math.max(1,avg))/6);
  const expectedGtin=Math.min(1,(gy*Math.max(1,avg))/10);
  const breadth=identityQueryBreadthScore(x.query||"");
  return 0.28*expectedGtin+0.34*expectedCanonical+0.18*volume+0.08*uy+0.07*Math.min(1,ce)+0.05*breadth+(runs===0?0.08:0);
}
function universalFulfillmentRoleScore(x={}){
  const dy=Number(x.history_decision_ready_yield||0),mty=Number(x.history_merchant_trust_yield||0),py=Number(x.history_purchase_ready_yield||0),uy=Number(x.history_unique_yield??0.5),runs=Number(x.history_runs||0);
  return UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_decision_weight*dy+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_trust_weight*mty+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_purchase_weight*py+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_unique_weight*uy+(runs===0?0.05:0);
}
function universalCrossProviderRoleScore(x={}){return 0.65*universalIdentityRoleScore(x)+0.35*universalFulfillmentRoleScore(x);}

function identityOptimizedSearchQuery(query="",exactId="",domain="general_commerce"){
  const exact=String(exactId||"").trim();
  if(exact&&validGtin(digitsOnly(exact)))return digitsOnly(exact);
  let q=String(query||"")
    .replace(/[※◆■●★☆]/g," ")
    .replace(/\b(?:new|used|sealed|open box|free shipping|parallel import|imported|returns?|refund|warranty|guarantee|cheap|sale)\b/gi," ")
    .replace(/(?:中古|新品|未開封|並行輸入品?|送料無料|返品(?:返金)?不可|保証(?:なし|無し)?|動作未確認|お安くご提供|商品として)/g," ")
    .replace(/\s+/g," ").trim();
  // Remove long disclaimer tails that commonly follow punctuation in marketplace titles.
  q=q.split(/[|｜;；]/)[0].trim();
  // Keep strong model / part tokens and enough surrounding product words for Yahoo identity recall.
  const tokens=q.split(/\s+/).filter(Boolean);
  const modelTokens=tokens.filter(t=>/[A-Za-z]/.test(t)&&/\d/.test(t)&&t.length>=4).slice(0,2);
  const base=[];
  for(const t of tokens){
    if(base.length>=8)break;
    if(t.length<=1)continue;
    base.push(t);
  }
  for(const t of modelTokens)if(!base.includes(t))base.push(t);
  const out=base.join(" ").trim();
  return (out||q||String(query||"")).slice(0,100);
}
function identityQueryBreadthScore(query=""){
  const q=String(query||"").trim(),tokens=q.split(/\s+/).filter(Boolean);
  const longPenalty=Math.max(0,tokens.length-10)*0.08+Math.max(0,q.length-100)/250;
  const identifierOnly=/^(?:GTIN|JAN|EAN|UPC|ISBN)?\s*\d{8,14}$/i.test(q)?0.55:0;
  return Math.max(0,1-longPenalty-identifierOnly);
}
