function purchaseDeadlineDays(deadline){
  if(!deadline)return null;const t=Date.parse(deadline);if(!Number.isFinite(t))return null;
  return Math.max(0,Math.ceil((t-Date.now())/86400000));
}
function purchaseKnownOfferTotal(offer,quantity=1){
  const price=purchaseFinite(offer?.price);if(price==null)return null;
  const shipping=purchaseFinite(offer?.shipping_cost);
  // A landed total is only known when shipping is explicitly known. Missing shipping is not zero.
  if(shipping==null)return null;
  return price*Math.max(1,quantity)+shipping;
}
function assessPartMerchantOffer(x,intent){
  const o=x?.offer||{},m=x?.merchant||{},score=x?.agent_score||{};
  const itemSubtotal=purchaseFinite(o?.price)==null?null:purchaseFinite(o.price)*Math.max(1,intent.quantity);
  const total=purchaseKnownOfferTotal(o,intent.quantity);
  const currency=String(o.currency||"").toUpperCase()||null;
  const max=intent.requirements.max_total_cost,deadlineDays=purchaseDeadlineDays(intent.requirements.delivery_deadline);
  const exact=normalizePartNumber(o.part_number||o.mpn||"")===normalizePartNumber(intent.product.mpn||"");
  const evidenceBackedAuthorized=Boolean(o.authorized_distributor&&(m.verified||o.authenticity_evidence_url));
  const authorizedOk=!intent.requirements.authorized_only||evidenceBackedAuthorized;
  const authenticityOk=!intent.requirements.authenticity_required||Boolean(m.verified||o.authenticity_evidence_url);
  // A hard budget cannot pass on a currency mismatch or unknown landed total.
  const budgetOk=max==null||(currency===intent.requirements.currency&&total!=null&&total<=max);
  // A hard delivery deadline cannot pass when lead time is unknown.
  const deliveryOk=deadlineDays==null||(o.lead_time_days!=null&&Number.isFinite(Number(o.lead_time_days))&&Number(o.lead_time_days)<=deadlineDays);
  const stockOk=["in_stock","low_stock"].includes(String(o.stock_status||""));
  const moq=purchaseFinite(o?.global_fulfillment?.minimum_order_quantity);
  const quantityOk=moq==null||intent.quantity>=moq;
  const ships=(o.ships_to||[]).map(v=>String(v).toUpperCase());
  const destinationOk=!ships.length||ships.includes(intent.destination.country)||ships.includes("WORLDWIDE")||ships.includes("ALL");
  const eligible=exact&&authorizedOk&&authenticityOk&&budgetOk&&deliveryOk&&stockOk&&quantityOk&&destinationOk;
  const unknown=[...(o.shipping_cost==null?["shipping"]:[])];
  if(o?.global_fulfillment?.tax_included!==true)unknown.push("tax_or_vat");
  if(o?.global_fulfillment?.duty_included!==true&&intent.destination.country!==String(m.country_code||"").toUpperCase())unknown.push("customs_or_duty");
  if(intent.destination.country!==String(m.country_code||"").toUpperCase())unknown.push("brokerage_if_applicable");
  return {...x,purchase_evaluation:{eligible,exact_identity:exact,authorized_requirement_met:authorizedOk,authenticity_requirement_met:authenticityOk,budget_requirement_met:budgetOk,delivery_requirement_met:deliveryOk,stock_usable:stockOk,quantity_requirement_met:quantityOk,destination_requirement_met:destinationOk,item_subtotal:itemSubtotal,known_total:total,currency,unknown_cost_components:[...new Set(unknown)],overall_agent_score:Number(score.overall_agent_score||0)}};
}
function globalPartsOfferBlockers(x,intent){
  const e=x?.purchase_evaluation||{},o=x?.offer||{};
  const blockers=[];
  if(!e.exact_identity)blockers.push("identity_mismatch");
  if(!e.stock_usable)blockers.push("stock_unavailable");
  if(!e.authorized_requirement_met)blockers.push("authorized_status_not_verified");
  if(!e.authenticity_requirement_met)blockers.push("authenticity_evidence_missing");
  if(!e.delivery_requirement_met)blockers.push(o.lead_time_days==null?"lead_time_unknown":"delivery_deadline_not_met");
  if(!e.budget_requirement_met)blockers.push(e.known_total==null?"landed_total_unknown_or_shipping_unknown":"budget_exceeded_or_currency_mismatch");
  if(!o.product_url&&!o.checkout_url)blockers.push("purchase_url_missing");
  return [...new Set(blockers)];
}
function globalPartsDecision(partIntel,evaluated,intent){
  const eligible=evaluated.filter(x=>x.purchase_evaluation?.eligible);
  const best=eligible[0]||null;
  const exact=evaluated.filter(x=>x.purchase_evaluation?.exact_identity);
  const stocked=exact.filter(x=>x.purchase_evaluation?.stock_usable);
  const successors=partIntel?.relations?.successors||[];
  const substitutes=partIntel?.relations?.verified_substitutes||[];
  let action="NO_VERIFIED_PURCHASE_ROUTE",reason_codes=[];
  if(best){action="BUY_EXACT";reason_codes=["exact_mpn","merchant_offer_eligible","hard_constraints_met","evidence_based_ranking"]; }
  else if(stocked.length){
    const blockers=[...new Set(stocked.flatMap(x=>globalPartsOfferBlockers(x,intent)))];
    if(blockers.some(x=>["authorized_status_not_verified","authenticity_evidence_missing"].includes(x))){action="VERIFY_SELLER_EVIDENCE";reason_codes=blockers;}
    else {action="VERIFY_FULFILLMENT_OR_TOTAL";reason_codes=blockers.length?blockers:["hard_constraint_unknown"]; }
  }else if(exact.length){action="WAIT_FOR_STOCK";reason_codes=["exact_offer_exists_but_not_currently_buyable"]; }
  else if(intent.requirements.substitutes_allowed&&successors.length){action="REVIEW_VERIFIED_SUCCESSOR";reason_codes=["no_eligible_exact_offer","verified_successor_available"]; }
  else if(intent.requirements.substitutes_allowed&&substitutes.length){action="REVIEW_VERIFIED_SUBSTITUTE";reason_codes=["no_eligible_exact_offer","verified_substitute_available"]; }
  else if(!intent.requirements.substitutes_allowed){action="NO_ELIGIBLE_EXACT_OFFER";reason_codes=["substitution_not_allowed","no_exact_offer_meets_constraints"]; }
  else {action="NO_SAFE_SUBSTITUTE";reason_codes=["no_exact_offer","no_verified_successor_or_substitute"]; }
  const evidenceComplete=best?(!globalPartsOfferBlockers(best,intent).length):false;
  const confidence=best&&evidenceComplete?"high":stocked.length?"medium":exact.length?"medium_low":"low";
  return {action,can_purchase:!!best,requires_user_confirmation:true,reason_codes,decision_confidence:confidence,best,eligible_count:eligible.length,exact_offer_count:exact.length,compatibility_policy:"never_infer_substitute_from_name_or_dimensions"};
}

function rankPurchasePartOffers(items=[]){
  return [...items].sort((a,b)=>{
    const A=a.purchase_evaluation||{},B=b.purchase_evaluation||{};
    if(Boolean(A.eligible)!==Boolean(B.eligible))return B.eligible?1:-1;
    const sa=Number(A.overall_agent_score||0),sb=Number(B.overall_agent_score||0);if(sa!==sb)return sb-sa;
    if(A.currency&&A.currency===B.currency&&A.known_total!=null&&B.known_total!=null)return A.known_total-B.known_total;
    return 0;
  });
}

async function bazaarComplianceAudit(origin,env){
  // v3.7.11: Do not use Coinbase /validate as the primary readiness gate.
  // That endpoint can return a uniform HTTP 429 because of Coinbase-side
  // validation throttling even when our seven paid resources are healthy.
  // Build the exact x402 v2 requirements locally from the same production
  // paymentRequirement() path used by the paid endpoints. This requires only
  // one facilitator /supported request for the entire seven-endpoint audit.
  const preflight=await x402DiscoveryPreflight(origin,env);
  const results=(preflight.results||[]).map(r=>({
    path:r.path,
    resource:`${origin}${r.path}`,
    http_status:r.ok?402:null,
    ok:!!r.ok,
    payment_required_header:!!r.payment_required_header,
    encoded_header_bytes:r.encoded_header_bytes??null,
    amount:r.amount??null,
    network:r.network??null,
    fee_payer_present:!!r.fee_payer_present,
    bazaar:!!r.bazaar,
    error:r.error||null
  }));
  const accepted=results.filter(r=>r.ok&&r.http_status===402&&r.payment_required_header&&r.bazaar);
  return {
    service:"ANIME INTELLIGENCE",
    version:VERSION,
    checked_at:new Date().toISOString(),
    audit_mode:"local_production_x402_requirement_preflight",
    external_coinbase_validate_called:false,

// v3.7.30 - keep the PAYMENT-REQUIRED header compact.
// Discovery metadata is intentionally rich in OpenAPI/MCP/.well-known/x402, but
// duplicating the full semantic agent-selection object twice inside the x402
// PaymentRequired extension can make the base64 PAYMENT-REQUIRED header very
// large. Some external directory probes fail at the transport/proxy layer before
// they can even observe HTTP 402 (reported as httpStatus:null / fetch failed).
// The payment handshake therefore carries only the canonical Bazaar fields
// required for discovery, while the full semantic metadata remains published on
// the dedicated discovery documents. Payment semantics and settlement are
// unchanged.
function bazaarPaymentExtension(path){
  return {
    info:{
      input:{
        type:"http",
        method:bazaarMethod(path),
        queryParams:bazaarExampleQueryParams(path),
        ...(bazaarExampleBody(path)?{body:bazaarExampleBody(path)}:{})
      },
      output:{
        type:"json",
        example:bazaarOutputExample(path)
      }
    },
    schema:bazaarInfoSchema(path)
  };
}

async function paymentRequirement(request,env,amount,description,supportOverride=null){
  if(!env.X402_WALLET_ADDRESS)throw new Error("X402_WALLET_ADDRESS is missing");
  const support=supportOverride||await facilitatorSupport(env);
  const kind=findSolanaMainnetKind(support);
  if(!kind)throw new Error("Facilitator does not advertise exact Solana mainnet");
  if(!kind.extra||!kind.extra.feePayer)throw new Error("Facilitator exact-SVM support is missing extra.feePayer");

  const path=new URL(request.url).pathname;
  const discovery=DISCOVERY_CONFIG[path]||{};
  const selector=agentSelectionMetadata(path);

  // Coinbase CDP x402 v2 rejects ResourceInfo.description when it is too long
  // and reports the misleading generic error that paymentPayload is invalid.
  // Keep the paid-response ResourceInfo compact; full discovery metadata stays
  // available through Bazaar extensions, OpenAPI, MCP and /.well-known/x402.
  const rawResourceDescription=[
    discovery.description||description,
    `WHEN TO USE: ${discovery.when_to_use||""}`,
    `WHY PAY: ${selector.why_pay||""}`,
    `BUYER OUTCOME: ${selector.buyer_outcome||""}`
  ].filter(Boolean).join(" ").replace(/\s+/g," ").trim();

  const displayPrice=endpointPriceUsd(path);
  const pricePrefix=Number.isFinite(Number(displayPrice))?`PRICE: ${Number(displayPrice)} USDC. `:"";
  const pricedResourceDescription=(pricePrefix+rawResourceDescription).trim();
  const resourceDescription=pricedResourceDescription.length>480
    ? pricedResourceDescription.slice(0,477)+"..."
    : pricedResourceDescription;

  const resource={
    url:canonicalPaidResourceUrl(request,path),
    description:resourceDescription,
    mimeType:"application/json",
    serviceName:BAZAAR_SERVICE_NAME,
    tags:bazaarTags(path),
    iconUrl:`${new URL(request.url).origin}/icon.svg`
  };

  const accepted={
    scheme:"exact",
    network:SOLANA_MAINNET,
    amount:String(amount),
    asset:SOLANA_USDC,
    payTo:String(env.X402_WALLET_ADDRESS),
    maxTimeoutSeconds:300,
    extra:{...kind.extra}
  };

  return {
    required:{
      x402Version:2,
      error:"PAYMENT-SIGNATURE header is required",
      resource,
      accepts:[accepted],
      extensions:{bazaar:bazaarPaymentExtension(path)}
    },
    accepted
  };
}

async function facilitatorPost(env,path,paymentPayload,accepted){
  const base=facilitatorUrl(env),u=new URL(`${base}${path}`),requestPath=u.pathname+u.search,auth=await facilitatorHeaders(env,"POST",requestPath);
  const r=await fetch(u.toString(),{method:"POST",headers:{...auth,"content-type":"application/json"},body:JSON.stringify({x402Version:2,paymentPayload,paymentRequirements:accepted})});
  const raw=await r.text();let body=null;
  try{body=raw?JSON.parse(raw):null;}catch{body={raw};}
  if(!r.ok)throw new Error(`Facilitator ${path} ${r.status}: ${JSON.stringify(body)}`);
  const extensionResponses=decodeExtensionResponsesHeader(r.headers.get("extension-responses"));
  if(body&&typeof body==="object"&&!Array.isArray(body))body.__extension_responses=extensionResponses;
  return body;
}

async function logRequestStage(env,request,eventType,extra={}){
  try{
    const url=new URL(request.url),base=requestTelemetry(request,url);
    // Privacy-preserving stable source grouping: no raw client IP is stored.
    const sourceMaterial=[base.source_class,base.user_agent,base.referer,base.origin,base.client_name,base.country].map(x=>String(x||"")).join("|");
    const sourceFingerprint=sourceMaterial.replace(/\|/g,"").trim()? (await sha256Hex(`anime-intelligence-client:${sourceMaterial}`)).slice(0,24):null;
    return await logEvent(env,eventType,{endpoint:url.pathname,product_id:extra.product_id||null,payer_hash:extra.payer_hash||null,amount_atomic:extra.amount_atomic??null,amount_usdc:extra.amount_usdc??null,payment_network:extra.payment_network||null,transaction_hash:extra.transaction_hash||null,metadata:{...base,source_fingerprint:sourceFingerprint,...(extra.metadata||{})}});
  }catch(e){return null;}
}

// v3.7.32 — commerce telemetry must never sit in front of the payment gate.
// In request handlers, Cloudflare ctx.waitUntil keeps KPI writes alive after the HTTP response
// without delaying 402/200 delivery. If ctx is unavailable, the promise is still safely caught.
function deferTelemetry(ctx,promise){
  const p=Promise.resolve(promise).catch(()=>null);
  try{if(ctx&&typeof ctx.waitUntil==="function")ctx.waitUntil(p);}catch{}
  return p;
}


async function paymentRequirementFingerprint(accepted){
  try{
    const stable={
      scheme:accepted?.scheme||null,
      network:accepted?.network||null,
      amount:String(accepted?.amount??""),
      asset:accepted?.asset||null,
      payTo:accepted?.payTo||null,
      feePayer:accepted?.extra?.feePayer||null
    };
    return (await sha256Hex(`x402-requirement:${JSON.stringify(stable)}`)).slice(0,32);
  }catch{return null;}
}
function paymentCapabilitySignals(request){
  const ua=String(request.headers.get("user-agent")||"").toLowerCase();
  const client=String(request.headers.get("x-client-name")||request.headers.get("x-agent-name")||"").toLowerCase();
  const accept=String(request.headers.get("accept")||"").toLowerCase();
  const paymentHeader=!!(request.headers.get("payment-signature")||request.headers.get("x-payment"));
  return {
    payment_header_present:paymentHeader,
    declares_x402:/x402|payment/.test(ua+" "+client+" "+accept),
    client_hint:(request.headers.get("x-client-name")||request.headers.get("x-agent-name")||null),
    user_agent_hint:String(request.headers.get("user-agent")||"").slice(0,180)||null
  };
}
