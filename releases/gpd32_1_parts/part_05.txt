function normalizePartNumber(v=""){
  return String(v||"").normalize("NFKC").toUpperCase().replace(/\s+/g,"").replace(/[‐‑‒–—―]/g,"-").trim();
}

function partRowShape(row){
  if(!row)return null;
  return {
    id:row.id,
    manufacturer:row.manufacturer||null,
    brand:row.brand||null,
    part_number:row.part_number||null,

    notes:row.notes||null,
    official_url:row.official_url||null,

    source_last_checked_at:row.source_last_checked_at||null,
    created_at:row.created_at||null,
    updated_at:row.updated_at||null
  };

function partSearchScore(row,q,pn,manufacturer){
  let score=0;
  const rpn=normalizePartNumber(row?.part_number||row?.normalized_part_number||"");
  const rq=normalize(String(q||""));
  const hay=normalize([row?.manufacturer,row?.brand,row?.part_number,row?.name,row?.category,row?.aliases_text].filter(Boolean).join(" "));
  if(pn&&rpn===normalizePartNumber(pn))score+=120;
  else if(pn&&rpn.includes(normalizePartNumber(pn)))score+=70;
  if(manufacturer&&normalize(row?.manufacturer||"")===normalize(manufacturer))score+=25;
  if(rq&&hay.includes(rq))score+=30;
  const toks=tokens(q||"").filter(Boolean);
  if(toks.length){
    const hit=toks.filter(t=>hay.includes(t)).length;
    score+=Math.min(30,hit*8);
  }
  score+=Math.min(10,Number(row?.source_confidence||0)*10);
  if(String(row?.lifecycle_status||"").toLowerCase()==="active")score+=3;
  return score;
}

async function searchPartsCatalog(env,{query="",part_number="",manufacturer="",limit=20}={}){
  const q=String(query||"").trim();
  const pn=normalizePartNumber(part_number||"");
  const m=String(manufacturer||"").trim();
  const lim=Math.max(1,Math.min(100,Number(limit)||20));
  let rows=[];

  if(pn){
    const encoded=encodeURIComponent(pn);
    const exact=await sbOptional(env,`/parts_catalog?select=*&normalized_part_number=eq.${encoded}&limit=${lim}`);
    if(Array.isArray(exact)&&exact.length)rows.push(...exact);
  }

  if(q||m){
    const terms=[];
    if(q){
      const safe=q.replace(/[%*,()]/g," ").replace(/\s+/g," ").trim().slice(0,120);
      if(safe){
        terms.push(`name.ilike.*${safe}*`);
        terms.push(`part_number.ilike.*${safe}*`);
        terms.push(`manufacturer.ilike.*${safe}*`);
        terms.push(`brand.ilike.*${safe}*`);
        terms.push(`category.ilike.*${safe}*`);
        terms.push(`aliases_text.ilike.*${safe}*`);
      }
    }
    if(m){
      const safeM=m.replace(/[%*,()]/g," ").replace(/\s+/g," ").trim().slice(0,100);
      terms.push(`manufacturer.ilike.*${safeM}*`);
      terms.push(`brand.ilike.*${safeM}*`);
    }
    if(terms.length){
      const found=await sbOptional(env,`/parts_catalog?select=*&or=${encodeURIComponent(`(${terms.join(",")})`)}&limit=${Math.max(lim,50)}`);
      if(Array.isArray(found))rows.push(...found);
    }
  }

  const uniq=new Map();
  for(const r of rows)if(r?.id&&!uniq.has(String(r.id)))uniq.set(String(r.id),r);
  const ranked=[...uniq.values()].map(r=>({row:r,score:partSearchScore(r,q,pn,m)})).sort((a,b)=>b.score-a.score);
  return ranked.slice(0,lim).map(x=>({...partRowShape(x.row),match_score:x.score}));
}

  }));
}

function relationBuckets(relations=[]){
  const keep=(t)=>relations.filter(r=>String(r.relation_type||"").toLowerCase()===t);
  return {
    successors:keep("successor"),
    verified_substitutes:relations.filter(r=>["substitute","compatible"].includes(String(r.relation_type||"").toLowerCase())&&r.verified&&Number(r.confidence)>=0.8),
    unverified_candidates:relations.filter(r=>["substitute","compatible"].includes(String(r.relation_type||"").toLowerCase())&&(!r.verified||Number(r.confidence)<0.8)),
    incompatible:keep("incompatible"),
    used_in:keep("used_in")
  };
}

function partDecision(part,buckets){
  const lifecycle=String(part?.lifecycle_status||"unknown").toLowerCase();
  if(buckets.successors.length)return {
    action:"CHECK_SUCCESSOR",
    reason:lifecycle==="discontinued"?"Original part is discontinued and a successor is recorded.":"A successor relationship is recorded.",
    target:buckets.successors[0].target_part
  };
  if(buckets.verified_substitutes.length)return {
    action:"VERIFIED_SUBSTITUTE_AVAILABLE",
    reason:"A verified substitute/compatibility relationship exists in the evidence table.",
    target:buckets.verified_substitutes[0].target_part
  };
  if(lifecycle==="discontinued")return {
    action:"SOURCE_ORIGINAL_OR_VERIFY_SUBSTITUTE",
    reason:"Part is marked discontinued and no verified successor/substitute is recorded.",
    target:null
  };
  return {
    action:"BUY_EXACT_OR_VERIFY_BEFORE_SUBSTITUTION",
    reason:"No verified compatibility relationship is recorded; title or dimensional similarity is not treated as compatibility proof.",
    target:null
  };
}

async function resolvePart(env,{query="",part_number="",manufacturer=""}={}){
  const results=await searchPartsCatalog(env,{query,part_number,manufacturer,limit:12});
  if(!results.length)return {ok:false,status:404,error:"part_not_found",results:[]};
  const first=results[0],second=results[1];
  const exact=part_number&&normalizePartNumber(first.part_number)===normalizePartNumber(part_number);
  const clearLead=!second||Number(first.match_score||0)-Number(second.match_score||0)>=15;
  if(!exact&&!clearLead){
    return {ok:false,status:409,error:"part_disambiguation_required",results:results.slice(0,8)};
  }
  return {ok:true,part:first,alternatives:results.slice(1,6),resolution:exact?"exact_part_number":"ranked_match"};
}

async function sha256HexText(v=""){
  const data=new TextEncoder().encode(String(v||""));
  const digest=await crypto.subtle.digest("SHA-256",data);
  return [...new Uint8Array(digest)].map(x=>x.toString(16).padStart(2,"0")).join("");
}

function merchantOfferScore(offer,merchant,buyerCountry=""){
  const o=merchantOfferShape(offer),m=merchantPublicShape(merchant)||{};
  let discover=0,trust=0,ready=0,compat=0,fulfill=0;
  const issues=[],improvements=[];
  if(o.manufacturer)discover+=15; else {issues.push("manufacturer_missing");improvements.push("Add the exact manufacturer name.");}
  if(o.part_number&&o.normalized_part_number)discover+=25; else {issues.push("mpn_missing");improvements.push("Add the exact manufacturer part number (MPN).");}
  if(o.merchant_sku)discover+=10;
  if(o.title)discover+=10; else improvements.push("Add a human-readable product title.");
  if(o.product_url)discover+=15; else {issues.push("product_url_missing");improvements.push("Add a stable product URL that an agent can hand off to.");}
  if(o.gtin)discover+=10; else improvements.push("Add GTIN/JAN/EAN when available.");
  if(o.datasheet_url)discover+=15; else improvements.push("Link the official datasheet or manufacturer document.");

  if(m.verified)trust+=30; else {issues.push("merchant_not_verified");improvements.push("Complete merchant verification.");}
  if(o.authorized_distributor){
    if(m.verified||o.authenticity_evidence_url)trust+=25;
    else {trust+=5;issues.push("authorized_status_unsubstantiated");improvements.push("Substantiate authorized-distributor status with merchant verification or supply-chain evidence.");}
  }else improvements.push("Declare authorized-distributor status only when it can be substantiated.");
  if(o.datasheet_url)trust+=15;
  if(o.authenticity_evidence_url)trust+=10; else improvements.push("Provide authenticity or supply-chain evidence when relevant.");
  if(o.return_policy_url)trust+=10; else {issues.push("return_policy_missing");improvements.push("Publish a machine-readable or stable return-policy URL.");}
  if(m.website_url)trust+=10;

  if(o.price!=null&&o.price>=0)ready+=25; else {issues.push("price_missing");improvements.push("Publish current price.");}
  if(o.currency)ready+=10; else improvements.push("Publish currency with price.");
  if(o.stock_status&&o.stock_status!=="unknown")ready+=20; else {issues.push("stock_unknown");improvements.push("Publish current stock status.");}
  if(o.stock_quantity!=null)ready+=10; else improvements.push("Publish stock quantity when operationally possible.");
  if(o.lead_time_days!=null)ready+=15; else {issues.push("lead_time_missing");improvements.push("Publish lead time in days.");}
  if(o.checkout_url||o.product_url)ready+=10;
  if(Array.isArray(o.ships_to)&&o.ships_to.length)ready+=10; else {issues.push("shipping_regions_missing");improvements.push("Publish ship-to country/region coverage.");}

  if(["in_stock","low_stock"].includes(o.stock_status))fulfill+=30;
  else if(o.stock_status==="backorder"||o.stock_status==="preorder")fulfill+=10;
  if(o.stock_quantity!=null&&o.stock_quantity>0)fulfill+=15;
  if(o.lead_time_days!=null){if(o.lead_time_days<=1)fulfill+=25;else if(o.lead_time_days<=3)fulfill+=20;else if(o.lead_time_days<=7)fulfill+=10;}
  const ships=(o.ships_to||[]).map(x=>String(x).toUpperCase());
  if(ships.length){if(!buyerCountry||ships.includes(String(buyerCountry).toUpperCase())||ships.includes("WORLDWIDE")||ships.includes("ALL"))fulfill+=20;}
  if(o.shipping_cost!=null)fulfill+=10;

  if(o.part_number&&o.mpn&&normalizePartNumber(o.part_number)===normalizePartNumber(o.mpn))compat+=40;
  if(o.datasheet_url)compat+=25;
  if(o.compatibility_evidence_url)compat+=25;
  if(o.compatibility_claim){
    if(o.compatibility_evidence_url)compat+=10;
    else {issues.push("compatibility_claim_without_evidence");improvements.push("A compatibility/substitution claim receives no ranking credit until an official evidence URL is supplied.");}
  }else if(!o.compatibility_evidence_url){
    improvements.push("If claiming substitution/compatibility, provide the official evidence URL.");
  }

  discover=clamp100(discover);trust=clamp100(trust);ready=clamp100(ready);compat=clamp100(compat);fulfill=clamp100(fulfill);
  const overall=clamp100(discover*.25+trust*.25+ready*.25+fulfill*.15+compat*.10);
  const band=overall>=85?"agent_ready":overall>=70?"competitive":overall>=50?"needs_improvement":"low_visibility";

function safeMerchantUrl(v){
  const s=String(v||"").trim();if(!s)return null;
  try{const u=new URL(s);if(!["http:","https:"].includes(u.protocol))return null;return u.toString();}catch{return null;}
}
async function merchantRegistrationIpHash(request){
  const raw=String(request?.headers?.get("cf-connecting-ip")||request?.headers?.get("x-forwarded-for")||"").split(",")[0].trim();
  return raw?await sha256HexText(`merchant-registration-v1|${raw}`):null;
}
async function enforceMerchantRegistrationRateLimit(env,request){
  const ipHash=await merchantRegistrationIpHash(request);
  if(!ipHash)return {ip_hash:null,recent_count:null};
  const cutoff=new Date(Date.now()-24*60*60*1000).toISOString();
  const path=`/merchants?select=id&metadata-%3E%3Eregistration_ip_hash=eq.${encodeURIComponent(ipHash)}&created_at=gte.${encodeURIComponent(cutoff)}&limit=4`;
  const rows=await sbOptional(env,path);
  const count=Array.isArray(rows)?rows.length:0;
  if(count>=3){
    const e=new Error("merchant_registration_rate_limited");
    e.status=429;
    throw e;
  }
  return {ip_hash:ipHash,recent_count:count};
}
function publicMerchantRegistrationBody(body){
  const name=String(body?.name||body?.merchant_name||"").trim().slice(0,120);
  if(name.length<2)throw new Error("merchant_name_too_short");

    country_code:country,
    verified:false,
    agent_description:String(body?.agent_description||"").trim().slice(0,1000)||null,

function merchantFiniteNumber(v,field,{min=null,max=null,integer=false}={}){
  if(v==null||v==="")return null;
  const n=Number(v);
  if(!Number.isFinite(n))throw new Error(`invalid_${field}`);
  const out=integer?Math.floor(n):n;
  if(min!=null&&out<min)throw new Error(`invalid_${field}`);
  if(max!=null&&out>max)throw new Error(`invalid_${field}`);
  return out;
}

async function searchMerchantOffers(env,{part_number="",manufacturer="",buyer_country="",limit=20}={}){
  const pn=normalizePartNumber(part_number||"");if(!pn)return {ok:false,status:400,error:"exact_part_number_required",count:0,offers:[]};

    scored.push({merchant:merchantPublicShape(merchant),offer:merchantOfferShape(row),agent_score:score});
  }
  scored.sort((a,b)=>Number(b.agent_score.overall_agent_score)-Number(a.agent_score.overall_agent_score)||((b.offer.stock_status==="in_stock")-(a.offer.stock_status==="in_stock"))||((a.offer.currency===b.offer.currency&&a.offer.price!=null&&b.offer.price!=null)?a.offer.price-b.offer.price:0));
