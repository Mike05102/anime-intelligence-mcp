function parseMonotaroOffers(source,page){
  const html=String(page.html||"");
  const text=String(page.text||stripHtml(html)).normalize("NFKC");
  const raw=html.normalize("NFKC");
  const decodedRaw=decodeEmbeddedUnicode(raw).normalize("NFKC");
  const pageUrl=String(page.official_url||page.url||"");
  if(/\/s\//.test(pageUrl))return [];
  if(!/\/p\//.test(pageUrl))return [];

  // Print pages are the canonical commerce evidence path for MonotaRO.
  // Use one pure parser for both production and self-test to prevent runtime drift.
  const combined=`${text}\n${stripHtml(decodedRaw)}`.normalize("NFKC");
  if(/\/print\/?(?:[?#].*)?$/.test(pageUrl)){
    const parsed=parseMonotaroEvidenceText(combined,pageUrl);
    return parsed?[parsed]:[];
  }

  // Normal product pages are identity/discovery fallback only. Keep fail-closed.
  const title=html?String(titleOf(html)||"").normalize("NFKC"):(text.match(/^Title:\s*(.+)$/mi)?.[1]||"");
  const explicitText=combined.match(/(?:\u30e1\u30fc\u30ab\u30fc)?\u54c1\u756a\s*[:\uff1a]?\s*([A-Z0-9][A-Z0-9._/+()\-]{1,100}(?:\s+\d+M)?)/i);
  const titleLead=title.match(/^\s*([A-Z0-9][A-Z0-9._/+()\-]{2,100}(?:\s+\d+M)?)(?=\s|$)/i);
  const displayPn=(explicitText?.[1]||titleLead?.[1]||"").trim().replace(/\s+/g," ");
  const normalizedPn=normalizePartNumber(displayPn);
  if(!normalizedPn || !/[A-Z]/.test(normalizedPn) || !/[0-9]/.test(normalizedPn))return [];
  const pathSku=pageUrl.match(/\/p\/(\d{4})\/(\d{4})\//)?.slice(1).join("")||null;
  return [{
    part_number:displayPn||normalizedPn,
    price:null,
    reference_price:null,
    availability_label:null,
    stock_status:"unknown",
    active:false,
    tax_included:false,
    lead_time_days:null,
    merchant_sku:pathSku,
    identity_evidence:explicitText?"visible_part_number":"document_title_exact_token",
    status_evidence:"normal_page_identity_only_fail_closed",
    commerce_window_chars:0,
    segment:""
  }];
}

async function ensureSystemMerchant(env,source){
  const found=await sbOptional(env,`/merchants?select=*&slug=eq.${encodeURIComponent(source.merchant_slug)}&limit=1`);
  if(Array.isArray(found)&&found.length)return found[0];
  const now=new Date().toISOString();
  // Deliberately discard the random preimage: system-managed merchants cannot
  // authenticate through the public merchant API unless explicitly converted later.
  const unusableSecret=`system-only-${crypto.randomUUID()}-${crypto.randomUUID()}`;
  const hash=await sha256HexText(unusableSecret);
  const row={
    id:crypto.randomUUID(),slug:source.merchant_slug,name:source.merchant_name,
    website_url:source.website_url,country_code:source.country_code||"JP",status:"active",
    verified:!!source.verified_merchant,merchant_api_key_hash:hash,merchant_api_key_prefix:"system:none",
    agent_description:`System-observed public offers from ${source.merchant_name}. No paid placement. Availability is represented conservatively.`,
    capabilities:{public_offer_observation:true,checkout_mode:"link_out"},
    metadata:{system_managed:true,source_type:source.source_type,source_url:source.website_url,merchant_self_service_auth:false},
    created_at:now,updated_at:now
  };
  const saved=await sb(env,"/merchants",{method:"POST",headers:{Prefer:"return=representation"},body:JSON.stringify(row)});
  const merchant=Array.isArray(saved)&&saved.length?saved[0]:row;
  await sbOptional(env,"/merchant_agent_profiles?on_conflict=merchant_id",{method:"POST",headers:{Prefer:"resolution=merge-duplicates,return=minimal"},body:JSON.stringify({
    merchant_id:merchant.id,discovery_name:source.merchant_name,
    description:`Public machine-observed offers from ${source.merchant_name}. Stock and fulfillment are never upgraded beyond the public evidence.`,
    checkout_mode:"link_out",supports_agent_checkout:false,protocols:["https"],
    capabilities:{public_offer_observation:true},policies:{ranking:"evidence_based_no_paid_placement",compatibility_claims_require_evidence:true},
    metadata:{system_managed:true,source_url:source.website_url},updated_at:now
  })});
  return merchant;
}

    :source.merchant_slug==="digikey-japan"?parseDigiKeyOffers(source,p)

  const originalUrl=String(item.url||"");

    try{
      const fallback=await fetchViaJinaOnly(originalUrl);

  return sbOptional(env,`/parts_collector_queue?id=eq.${encodeURIComponent(item.id)}`,{method:"PATCH",body:JSON.stringify({status:"done",processed_at:new Date().toISOString(),last_error:null,metadata:result,updated_at:new Date().toISOString()})});
}
async function markFailure(env,item,e){
  const attempts=Number(item.attempts||0)+1;
  const error=safeError(e);
  const policy=retryPolicy(error,attempts);
  const next=new Date(Date.now()+policy.delay_hours*3600000).toISOString();
  return sbOptional(env,`/parts_collector_queue?id=eq.${encodeURIComponent(item.id)}`,{method:"PATCH",body:JSON.stringify({status:policy.status,attempts,next_attempt_at:next,last_error:`${policy.reason}:${error}`,updated_at:new Date().toISOString()})});
}

async function bootstrapDeepWatchWindow(env,window){
  const now=new Date().toISOString();
  for(const source of (window||[]).slice(0,Math.max(1,DEEP_BOOTSTRAP_PER_RUN))){
    const seed=(source.seeds||[])[0];
    if(!seed)continue;
    const got=await sbOptional(env,`/parts_collector_queue?select=*&source_slug=eq.${encodeURIComponent(source.slug)}&url=eq.${encodeURIComponent(seed)}&limit=1`);
    const existing=Array.isArray(got)&&got.length?got[0]:null;
    if(existing){
      if(isDeepWatchDepth(existing.depth))continue;
      const row={...existing,status:"pending",depth:-1,attempts:0,next_attempt_at:now,last_error:null,updated_at:now};
      await sbOptional(env,`/parts_collector_queue?id=eq.${encodeURIComponent(existing.id)}`,{method:"PATCH",headers:{Prefer:"return=minimal"},body:JSON.stringify({status:"pending",depth:-1,attempts:0,next_attempt_at:now,last_error:null,updated_at:now})});
      return {used:true,source_slug:source.slug,url:seed,row};
    }
    const row={id:crypto.randomUUID(),source_slug:source.slug,url:seed,status:"pending",depth:-1,attempts:0,next_attempt_at:now,created_at:now,updated_at:now};
    await sbOptional(env,"/parts_collector_queue?on_conflict=source_slug,url",{method:"POST",headers:{Prefer:"resolution=ignore-duplicates,return=minimal"},body:JSON.stringify(row)});
    return {used:true,source_slug:source.slug,url:seed,row};
  }
  return {used:false};
}

async function refreshStaleWatchRows(env){
  const enabled=OFFICIAL_WATCH_SOURCES.filter(x=>x.collection_enabled!==false);
  if(!enabled.length)return 0;
  const slugs=enabled.map(x=>x.slug).join(",");
  const cutoff=new Date(Date.now()-7*86400000).toISOString();
  const rows=await sbOptional(env,`/parts_collector_queue?select=id,source_slug,status,processed_at&source_slug=in.(${encodeURIComponent(slugs)})&status=eq.done&processed_at=lt.${encodeURIComponent(cutoff)}&order=processed_at.asc&limit=8`);
  let n=0;
  for(const r of Array.isArray(rows)?rows:[]){await sbOptional(env,`/parts_collector_queue?id=eq.${encodeURIComponent(r.id)}`,{method:"PATCH",body:JSON.stringify({status:"pending",next_attempt_at:new Date().toISOString(),updated_at:new Date().toISOString()})});n++;}
  return n;
}

const RELATION_STATUS_DIAGNOSTIC_PATH="/parts_relation_candidates?select=id,verified,source_part_number,target_part_number,compatibility_warning,parser_confidence,observed_at&limit=60";

async function diagnosticQuery(env,name,path,timeoutMs=SUPABASE_TIMEOUT_MS,fallbackPath=null){
  try{
    const data=await sb(env,path,{timeout_ms:timeoutMs,retry_get:false});
    return {name,ok:true,sample_ok:true,data,error:null,fallback_used:false};
  }catch(e){
    const primaryError=safeError(e);
    if(fallbackPath){
      try{
        const fallback=await sb(env,fallbackPath,{timeout_ms:STATUS_FALLBACK_TIMEOUT_MS,retry_get:false});
        return {name,ok:true,sample_ok:false,data:null,error:primaryError,fallback_used:true,fallback_data:fallback};
      }catch(fe){
        return {name,ok:false,sample_ok:false,data:null,error:primaryError,fallback_used:true,fallback_error:safeError(fe)};
      }
    }
    return {name,ok:false,sample_ok:false,data:null,error:primaryError,fallback_used:false};
  }
}
async function statusDbProbe(env){
  const path="/parts_ingest_runs?select=version,finished_at&limit=1";
  let first=await diagnosticQuery(env,"probe",path,STATUS_PROBE_TIMEOUT_MS,"/parts_ingest_runs?select=version&limit=1");
  if(first.ok&&first.sample_ok)return {...first,retried:false};
  await new Promise(r=>setTimeout(r,250));
  const second=await diagnosticQuery(env,"probe",path,STATUS_RETRY_TIMEOUT_MS,"/parts_ingest_runs?select=version&limit=1");
  return {...second,retried:true,first_error:first.error||null};
}
async function runDiagnosticBatches(env,specs){
  const out=[];
  for(let i=0;i<specs.length;i+=STATUS_DIAGNOSTIC_BATCH_SIZE){
    const batch=specs.slice(i,i+STATUS_DIAGNOSTIC_BATCH_SIZE);
    const rows=await Promise.all(batch.map(x=>diagnosticQuery(env,x.name,x.path,x.timeout_ms||SUPABASE_TIMEOUT_MS,x.fallback_path||null)));
    out.push(...rows);
  }
  return out;
}

// v7.1 deterministic cron contract. Works with either */1 or */10 Cloudflare triggers.
// Only UTC minutes divisible by 10 perform network-heavy scheduled work.
// 00/20/40 = embedded Parts Collector v3; 10 = universal rotation;
// 30 = catalog enrichment; 50 = universal demand refresh.
function scheduledTaskForTime(timeMs){
  // All ten-minute slots are productive. This avoids relying on :40/:50 cron configuration.
  const d=new Date(Number(timeMs||Date.now()));
  const m=d.getUTCMinutes();
  if(m%10!==0)return "discovery_recovery"; // also works with an unusual single-minute cron
  if(m===0||m===20)return "parts_collector_v3";
  if(m===10)return "rotation";
  if(m===30)return "catalog_enrichment";
  if(m===40)return "universal_discovery";
  if(m===50)return "universal_discovery";
  return "discovery_recovery";
}
