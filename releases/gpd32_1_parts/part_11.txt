async function sha256HexText(v=""){
  const b=new TextEncoder().encode(String(v));
  const h=await crypto.subtle.digest("SHA-256",b);
  return [...new Uint8Array(h)].map(x=>x.toString(16).padStart(2,"0")).join("");
}

}

function isRetryableSupabaseReadError(e){
  const x=String(e?.message||e||"");
  return /Supabase timeout|Supabase (503|504|520)|PGRST00[0-3]|network|fetch failed|connection/i.test(x);

async function sbOnce(env,path,options={}){
  const controller=new AbortController();
  const timeoutMs=Math.max(3000,Math.min(30000,Number(options.timeout_ms||SUPABASE_TIMEOUT_MS)));
  const timer=setTimeout(()=>controller.abort("supabase_timeout"),timeoutMs);
  try{
    const r=await fetch(sbBase(env)+path,{method:options.method||"GET",headers:sbHeaders(env,options.headers||{}),body:options.body,signal:controller.signal});
    const raw=await r.text();let data=null;try{data=raw?JSON.parse(raw):null;}catch{data={raw};}
    if(!r.ok)throw new Error(`Supabase ${r.status}: ${raw.slice(0,800)}`);
    return data;
  }catch(e){
    if(String(e?.name||"")==="AbortError")throw new Error(`Supabase timeout after ${timeoutMs}ms: ${path.slice(0,220)}`);
    throw e;
  }finally{clearTimeout(timer);}
}
async function sb(env,path,options={}){
  const method=String(options.method||"GET").toUpperCase();
  const retries=supabaseReadRetryCount(method,options.retry_get!==false);
  let lastErr=null;
  for(let attempt=0;attempt<=retries;attempt++){
    try{
      const timeoutMs=attempt===0?options.timeout_ms:Math.min(30000,Number(options.timeout_ms||SUPABASE_TIMEOUT_MS)+5000);
      return await sbOnce(env,path,{...options,timeout_ms:timeoutMs});
    }catch(e){
      lastErr=e;
      if(attempt>=retries||!isRetryableSupabaseReadError(e))throw e;
      await new Promise(r=>setTimeout(r,SUPABASE_GET_RETRY_DELAY_MS*(attempt+1)));
    }
  }
  throw lastErr||new Error("supabase_unknown_error");
}

function rotateWindow(items,tick,width){
  if(!Array.isArray(items)||!items.length)return [];
  const w=Math.max(1,Math.min(items.length,Number(width)||1));
  const start=((Number(tick)||0)*w)%items.length;
  const out=[];
  for(let i=0;i<w;i++)out.push(items[(start+i)%items.length]);
  return out;
}

function merchantEvidenceGate(source,offer,pageText="",pageUrl=""){
  const pn=normalizePartNumber(offer?.part_number||"");
  if(!pn)return {ok:false,reason:"missing_mpn"};
  const hay=`${String(pageText||"")} ${String(pageUrl||"")}`.normalize("NFKC").toUpperCase();
  const compactHay=normalizePartNumber(hay);
  if(!compactHay.includes(pn))return {ok:false,reason:"mpn_not_visible"};
  if(source?.generated_adapter===true && !isLikelyMerchantDetailUrl(source,pageUrl))return {ok:false,reason:"generated_adapter_search_page_not_evidence"};
  if(source?.generated_adapter===true){
    const brandOk=manufacturerAliases(source.manufacturer).some(a=>hay.includes(a));
    if(!brandOk)return {ok:false,reason:"manufacturer_not_corroborated"};
  }
  const hasPositiveCommerce=offer?.active===true || offer?.active===false || offer?.price!=null || offer?.stock_status && offer.stock_status!=="unknown" || !!offer?.status_evidence;
  if(!hasPositiveCommerce)return {ok:false,reason:"no_commerce_evidence"};
  return {ok:true,reason:"evidence_gate_pass"};
}
function retryPolicy(error,attempts){
  const e=String(error||"");
  if(/HTTP_429|JINA_HTTP_429/.test(e))return {status:attempts>=6?"failed":"retry",delay_hours:24,reason:"rate_limited"};
  if(/HTTP_403/.test(e))return {status:attempts>=4?"failed":"retry",delay_hours:72,reason:"blocked"};
  if(/Supabase 504|Supabase timeout|upstream request timeout/i.test(e))return {status:attempts>=6?"failed":"retry",delay_hours:1,reason:"database_timeout"};
  const h=Math.min(24,Math.pow(2,Math.max(1,attempts)));
  return {status:attempts>=6?"failed":"retry",delay_hours:h,reason:"exponential"};
}



function decodeHttpBodyBytes(bytes,contentType=""){
  const labels=[];
  const m=String(contentType||"").match(/charset\s*=\s*["']?([^;"'\s]+)/i);
  if(m?.[1])labels.push(m[1].toLowerCase());
  for(const x of ["utf-8","shift_jis"])if(!labels.includes(x))labels.push(x);
  const candidates=[];
  for(const label of labels){
    try{
      const t=new TextDecoder(label,{fatal:false}).decode(bytes);
      candidates.push({label,text:t,score:textDecodeScore(t)});
    }catch{}
  }
  if(!candidates.length)return {text:"",charset:"unknown"};
  candidates.sort((a,b)=>b.score-a.score);
  return {text:candidates[0].text,charset:candidates[0].label};
}
function textDecodeScore(text=""){
  const s=String(text);
  let jp=0,bad=0;
  for(const ch of s){
    const c=ch.codePointAt(0);
    if((c>=0x3040&&c<=0x30ff)||(c>=0x3400&&c<=0x9fff)||(c>=0xff00&&c<=0xffef))jp++;
    if(ch==="\ufffd")bad+=8;
  }
  const moj=(s.match(/[\u00c3\u00c2\u00e2\u00e5\u00e3]/g)||[]).length;
  return jp*3-bad-moj*2;
}

async function fetchPage(url){
  // PDFs are retained as first-party evidence pointers instead of repeatedly
  // fetching binary bodies through Cloudflare/Jina. This avoids 403->429 retry
  // loops while preserving the official manufacturer URL for provenance.
  if(isPdfUrl(url)){
    return {
      url,
      official_url:url,
      isPdf:true,
      html:"",
      text:"",
      status:200,
      transport:"official_pdf_pointer",
      body_fetched:false,
      pointer_only:true,
      note:"Official PDF URL retained as evidence pointer; body parsing deferred."
    };
  }

  let directError=null;
  try{
    const r=await fetch(url,{headers:crawlerHeaders(),redirect:"follow"});
    const ct=String(r.headers.get("content-type")||"").toLowerCase();
    if(r.ok && !ct.includes("application/pdf")){
      const bytes=new Uint8Array(await r.arrayBuffer());
      const decoded=decodeHttpBodyBytes(bytes,ct);
      const html=decoded.text.slice(0,MAX_BODY_CHARS);
      return {url:r.url||url,official_url:url,isPdf:false,html,text:stripHtml(html),status:r.status,transport:"direct",body_fetched:true,decoded_charset:decoded.charset};
    }
    if(!r.ok)directError=`HTTP_${r.status}`;
    else if(ct.includes("application/pdf")){
      return {
        url:r.url||url,
        official_url:url,
        isPdf:true,
        html:"",
        text:"",
        status:r.status,
        transport:"official_pdf_pointer",
        body_fetched:false,
        pointer_only:true,
        note:"Official endpoint returned PDF; body parsing deferred."
      };
    }
  }catch(e){directError=safeError(e);}

  // HTML/text fallback transport only. Provenance remains the original
  // manufacturer URL. Jina is never used for PDF bodies in v1.3.2.
  try{
    const jr=await fetch(jinaReaderUrl(url),{headers:{"accept":"text/plain,text/markdown;q=0.9,*/*;q=0.8"},redirect:"follow"});
    if(!jr.ok)throw new Error(`JINA_HTTP_${jr.status}`);
    const body=(await jr.text()).slice(0,MAX_BODY_CHARS);
    return {url,official_url:url,isPdf:false,html:"",text:body,status:200,transport:"jina_reader",direct_error:directError,body_fetched:true};
  }catch(e){
    throw new Error(`${directError||"DIRECT_FETCH_FAILED"};FALLBACK_${safeError(e)}:${url}`);
  }
}

}

  if(manufacturer==="Mitsubishi Electric")return [/\b(?:QJ71|Q\d{2,4}|FX1|FX2|FX3|FX5|R\d{2,4}|RJ71|RD\d{2,4}|MR-J2|MR-J3|MR-J4|MR-J5|MR-JE|HG-[A-Z]|HF-[A-Z]|FR-[A-Z]|GT\d{3,5}|AJ65|A1S|A2S)[A-Z0-9\-\/\.]{0,30}\b/gi];
  if(manufacturer==="SMC")return [/\b(?:SY3|SY5|SY7|VQ|CQ2|CDQ2|C85|C96|MGPM|MXS|MXQ|KQ2|AS\d|AR\d|AW\d|ZSE|ISE|LE[A-Z]|PFM|PF3W)[A-Z0-9\-\/\.]{1,32}\b/gi];
  if(manufacturer==="Panasonic Industry")return [/\b(?:FX-|CX-|EX-|EQ-|PM-|HG-|AFP|FP0R|FP-X|GT\d|A5|A6)[A-Z0-9\-\/\.]{1,30}\b/gi];
  if(manufacturer==="Yaskawa")return [/\b(?:SGDM|SGDH|SGDS|SGDV|SGD7S|SGM[A-Z]|SGLC|SGT[A-Z0-9]|CACR|CPCR|JUSP|SJDE|CEPT|SSR|SST|EMR|EMT|EMG|DR[A-Z0-9])[A-Z0-9\-\/\.]{0,32}\b/gi];
  if(manufacturer==="THK")return [/\b(?:HSR|SRS|SRG|RSR|SBN|BNS|LTR|FBL|BTK|KSH|CCM|LXA|LXT|LM(?:E|JK|R)?|LBS)[A-Z0-9\-\/\.]{0,28}\b/gi];
  if(manufacturer==="IDEC")return [/\b(?:HW1|HW4|XW1E|XW5E|HS5D|FC6A|FT1A|KW2D|PS5R|RJ|RU|RH|LBW|AP22|BN)[A-Z0-9\-\/\.]{1,28}\b/gi];
  if(manufacturer==="Oriental Motor")return [/\b(?:PKP|AZX|AZ|BLM|BXM|MSD|MACK|MAC-|EP|EPUA|EPR|CS[25]|CC20|PAES|PAVR|FD|FGB|LXD)[A-Z0-9\-\/\.]{1,32}\b/gi];

  return [];
}
function genericModels(text=""){
  const out=[];
  const rx=/\b(?=[A-Z0-9\-\/\.]{5,30}\b)(?=[A-Z0-9\-\/\.]*[A-Z])(?=[A-Z0-9\-\/\.]*\d)[A-Z][A-Z0-9]{0,8}(?:[-\/\.][A-Z0-9]{1,12}){0,4}\b/g;
  for(const m of String(text).toUpperCase().matchAll(rx))out.push(m[0]);
  return out;
}
function extractModels(manufacturer,text){
  const out=[];
  const raw=String(text||"").normalize("NFKC");
  for(const rx of modelPatterns(manufacturer)){rx.lastIndex=0;for(const m of raw.matchAll(rx))out.push(m[0]);}
  for(const x of genericModels(raw))out.push(x);
  const bad=/^(?:HTTPS?|HTML|UTF-8|JAPAN|DOWNLOAD|PRODUCT|PRODUCTS|CATALOG|MANUAL|SUPPORT|CONTACT|COPYRIGHT|WIDTH|HEIGHT|PAGE|PDF)$/i;
  return [...new Set(out.map(normalizePartNumber).filter(x=>x.length>=4&&x.length<=34&&!bad.test(x)))].slice(0,MAX_MODELS_PER_PAGE);
}

function occurrenceCount(text,token){
  const t=String(text||"").toUpperCase(),q=String(token||"").toUpperCase();
  if(!q)return 0;let n=0,pos=0;while((pos=t.indexOf(q,pos))>=0){n++;pos+=q.length;if(n>=4)break;}return n;
}
function contextualOfficialModels(manufacturer,text,title=""){
  const raw=String(`${title||""} ${text||""}`).normalize("NFKC");
  const specific=[];
  for(const rx of modelPatterns(manufacturer)){rx.lastIndex=0;for(const m of raw.matchAll(rx))specific.push(normalizePartNumber(m[0]));}
  const contextCue=/(?:品番|型番|形式|形名|製品番号|モデル|model|part\s*(?:no\.?|number)|mfr\.?\s*part|order\s*code|product\s*code)/i;
  const generic=genericModels(raw).map(normalizePartNumber).filter(pn=>{
    if(pn.length<5||pn.length>34)return false;
    if(occurrenceCount(raw,pn)>=2)return true;
    const i=raw.toUpperCase().indexOf(pn.toUpperCase());
    if(i<0)return false;
    return contextCue.test(raw.slice(Math.max(0,i-90),Math.min(raw.length,i+pn.length+90)));
  });
  return [...new Set([...specific,...generic])].slice(0,WATCH_ENRICHMENT_MAX_PARTS);
}
function metricExcerpt(text,index,len=120){const t=String(text||"").replace(/\s+/g," ");return t.slice(Math.max(0,index-45),Math.min(t.length,index+len)).trim();}
function findMetricAfterLabels(text,labels,units,sourceUrl,key){
  const t=String(text||"").normalize("NFKC");
  const lower=t.toLowerCase();
  for(const label of labels){
    const ll=String(label).normalize("NFKC").toLowerCase();
    let pos=0;
    while((pos=lower.indexOf(ll,pos))>=0){
      const tail=t.slice(pos+ll.length,pos+ll.length+80).replace(/^\s*[:：]?\s*/,"");
      const nm=tail.match(/^-?\d+(?:\.\d+)?/);
      if(nm){
        const value=Number(nm[0]);
        const after=tail.slice(nm[0].length).trimStart();
        for(const u of units){
          const nu=String(u).normalize("NFKC");
          if(after.toLowerCase().startsWith(nu.toLowerCase())) return {key,value,unit:u,source_url:sourceUrl,evidence:metricExcerpt(t,pos)};
        }
      }
      pos+=Math.max(1,ll.length);
    }
  }
  return null;
}

function canonicalEvidenceForModel(source,text,title,url,pn){
  const raw=String(`${title||""} ${text||""}`).normalize("NFKC");
  const norm=normalizePartNumber(pn);
  const upper=raw.toUpperCase();
  const idx=upper.indexOf(norm);
  const explicitCue=/(?:品番|型番|形式|形名|製品番号|モデル|model|part\s*(?:no\.?|number)|mfr\.?\s*part|order\s*code|product\s*code)/i;
  const detailUrl=/(?:\/detail(?:\/|$)|\/product(?:\/|$)|\/products\/[^/?#]+(?:\/[^/?#]+)?$|\/p\/|\/item\/|\/model\/|datasheet|spec(?:ification)?)/i.test(String(url||""));
  const titleHit=normalizePartNumber(title||"").includes(norm);
  const context=idx>=0?raw.slice(Math.max(0,idx-120),Math.min(raw.length,idx+norm.length+120)):"";
  const specific=modelPatterns(source.manufacturer).some(rx=>{rx.lastIndex=0;return rx.test(norm);});
  if((detailUrl&&(titleHit||explicitCue.test(context))) || (titleHit&&explicitCue.test(raw)))return {level:"manufacturer_exact",reason:"official_detail_identity",confidence:0.92};
  if(specific && (occurrenceCount(raw,norm)>=2 || explicitCue.test(context)))return {level:"manufacturer_catalog",reason:"official_catalog_identity",confidence:0.80};
  return {level:"candidate",reason:"weak_or_pagewide_identifier",confidence:0.50};
}

function rankOfficialDiscoveryUrl(url=""){
  const u=String(url||"");let score=0;
  if(/replacement|successor|renew|discontinued/i.test(u))score+=8;
  if(/\.pdf(?:$|[?#])/i.test(u))score+=6;
  if(/\/detail|\/item|\/model|datasheet|spec/i.test(u))score+=5;
  if(/[?&](?:page|p|sort|filter|rv|si|pf|pt|attr_)=/i.test(u))score-=8;
  return score;
}

async function deepQueueBacklog(env){
  const rows=await sbOptional(env,`/parts_collector_queue?select=id&source_slug=like.watch-*&status=in.(pending,retry)&depth=lt.0&limit=${DEEP_QUEUE_BUDGET+1}`);

}
async function enqueueDeepWatchLinks(env,source,links,itemDepth){
  const hop=isDeepWatchDepth(itemDepth)?deepHop(itemDepth)+1:1;
  if(hop>MAX_DEEP_HOPS)return 0;
  const backlog=await deepQueueBacklog(env);if(backlog!=null&&backlog>=DEEP_QUEUE_BUDGET)return 0;
  const clean=[...new Set((links||[]).filter(u=>pageAllowed(source,u)&&detailLikeOfficialUrl(u)))].slice(0,Math.max(1,DEEP_WATCH_PRIORITY_LIMIT*2));
  if(!clean.length)return 0;
  const now=new Date().toISOString(),depth=-hop;
  const rows=clean.map(u=>({id:crypto.randomUUID(),source_slug:source.slug,url:u,status:"pending",depth,attempts:0,next_attempt_at:now,created_at:now,updated_at:now}));
  await sbOptional(env,"/parts_collector_queue?on_conflict=source_slug,url",{method:"POST",headers:{Prefer:"resolution=ignore-duplicates,return=minimal"},body:JSON.stringify(rows)});
  return rows.length;
}

function merchantDetailCandidates(source,page){
  const links=allLinksOf(page).filter(u=>pageAllowed(source,u));
  const exact=links.filter(u=>isMerchantDetailUrl(source,u));
  if(exact.length)return [...new Set(exact)].slice(0,MERCHANT_DETAIL_PROMOTION_LIMIT);
  // MonotaRO search/category pages usually lead to /g/ family pages first; those
  // family pages then expose exact /p/ SKU pages. Treat /g/ only as discovery,
  // never as final offer evidence.
  if(source?.merchant_slug==="monotaro-japan")return [...new Set(links.filter(u=>isMerchantDiscoveryHubUrl(source,u)))].slice(0,MERCHANT_DETAIL_PROMOTION_LIMIT);
  return [];
}

function strongRelationPairs(source,text,url,title=""){
  const t=String(text||"").normalize("NFKC");
  const pairs=[];
  const compatibilityWarning=/\u76f4\u63a5(?:\u306e)?\u4e92\u63db(?:\u5546\u54c1|\u6a5f\u7a2e)?\u3067\u306a\u3044|\u4e92\u63db\u6027(?:\u304c)?\u306a\u3044|\u305d\u306e\u307e\u307e(?:\u3067\u306f)?\u7f6e\u304d\u63db\u3048\u3067\u304d\u306a\u3044|\u914d\u7dda\u63a5\u7d9a.*\u5909\u308f|\u53d6\u4ed8\u5bf8\u6cd5.*\u5909\u308f|\u5916\u5f62\u5bf8\u6cd5.*\u5909\u308f/i.test(t);
  const incompatNotes=[];
  if(/\u5916\u5f62\u5bf8\u6cd5.*\u5909\u308f|\u5916\u5f62\u5bf8\u6cd5.*\u00d7/i.test(t))incompatNotes.push("dimensions_change");
  if(/\u914d\u7dda\u63a5\u7d9a.*\u5909\u308f|\u914d\u7dda\u63a5\u7d9a.*\u00d7|\u7aef\u5b50\u914d\u7f6e.*\u5909\u308f/i.test(t))incompatNotes.push("wiring_change");
  if(/\u53d6\u4ed8\u5bf8\u6cd5.*\u5909\u308f|\u53d6\u4ed8\u5bf8\u6cd5.*\u00d7/i.test(t))incompatNotes.push("mounting_change");

  // Strongest parser: official replacement-table rows. Many OMRON PDFs render one source/target pair per line.
  const lines=t.split(/\r?\n/).map(x=>x.trim()).filter(Boolean);
  for(const line of lines){
    if(!/(\u63a8\u5968\u4ee3\u66ff\u5546\u54c1|\u63a8\u5968\u4ee3\u66ff\u54c1)/i.test(t))break;
    const ms=uniqueModels(source.manufacturer,line);
    if(/\u63a8\u5968\u4ee3\u66ff\u5546\u54c1(?:\u306f)?(?:\u3042\u308a\u307e\u305b\u3093|\u306a\u3057)/i.test(line)&&ms.length){
      for(const a of ms.slice(0,4))pairs.push({part_number:a,source_part_number:a,target_part_number:null,relation_hint:"no_recommended_replacement",relation_type_candidate:null,source_url:url,source_title:title||null,evidence_type:"official_no_recommended_replacement",evidence_excerpt:compactEvidence(line),parser_confidence:0.99,compatibility_warning:false,incompatibility_notes:[],verified:false});
      continue;
    }
    if(ms.length>=2){
      const a=normalizePartNumber(ms[0]),b=normalizePartNumber(ms[1]);
      if(a&&b&&a!==b){
        pairs.push({part_number:a,source_part_number:a,target_part_number:b,relation_hint:"successor_or_substitute_candidate",relation_type_candidate:"substitute",source_url:url,source_title:title||null,evidence_type:"official_replacement_table_row",evidence_excerpt:compactEvidence(line),parser_confidence:0.99,compatibility_warning:compatibilityWarning,incompatibility_notes:incompatNotes,verified:false});
      }
    }
    if(pairs.length>=60)break;
  }
  if(pairs.some(p=>p.evidence_type==="official_replacement_table_row"||p.evidence_type==="official_no_recommended_replacement")){
    const seen=new Set();
    return pairs.filter(p=>{const k=`${p.source_part_number}|${p.target_part_number||"NONE"}|${p.source_url}|${p.relation_hint}`;if(seen.has(k))return false;seen.add(k);return true;});
  }

  const markers=[...t.matchAll(/\u63a8\u5968\u4ee3\u66ff\u5546\u54c1|\u63a8\u5968\u4ee3\u66ff\u54c1|\u4ee3\u66ff\u6a5f\u7a2e|\u7f6e\u304d\u63db\u3048\u6a5f\u7a2e|\u7f6e\u63db\u6a5f\u7a2e|\u5f8c\u7d99\u6a5f\u7a2e/gi)];
  for(const m of markers.slice(0,12)){
    const idx=m.index||0;
    const before=t.slice(Math.max(0,idx-1200),idx);
    const after=t.slice(idx,Math.min(t.length,idx+1800));
    const sourceSection=(before.match(/(?:\u751f\u7523\u7d42\u4e86\u5546\u54c1|\u751f\u7523\u7d42\u4e86\u54c1|\u5f93\u6765\u6a5f\u7a2e|\u65e7\u6a5f\u7a2e)([\s\S]*)$/i)||[])[1]||before.slice(-650);
    const targetSection=after.split(/(?:\u25a0|\u6ce8\u610f|\u76f8\u9055\u70b9|\u6700\u7d42\u53d7\u6ce8|\u6700\u7d42\u51fa\u8377|\u751f\u7523\u7d42\u4e86\u5546\u54c1|\u5f93\u6765\u6a5f\u7a2e)/i)[0];
    const src=uniqueModels(source.manufacturer,sourceSection).slice(0,8);
    const dst=uniqueModels(source.manufacturer,targetSection).filter(x=>!src.includes(x)).slice(0,8);
    if(!src.length||!dst.length)continue;
    const confidence=(src.length===1&&dst.length===1)?0.97:(src.length<=2&&dst.length<=3)?0.90:0.78;
    for(const a of src)for(const b of dst){
      pairs.push({
        part_number:a,source_part_number:a,target_part_number:b,
        relation_hint:"successor_or_substitute_candidate",
        relation_type_candidate:"substitute",
        source_url:url,source_title:title||null,
        evidence_type:"official_replacement_section",
        evidence_excerpt:compactEvidence(`${sourceSection} ${targetSection}`),
        parser_confidence:confidence,
        compatibility_warning:compatibilityWarning,
        incompatibility_notes:incompatNotes,
        verified:false
      });
      if(pairs.length>=40)return pairs;
    }
  }

  // Mitsubishi and other official pages often use compact 'A \u2192 B' / 'A\u304b\u3089B\u3078' style wording.
  const arrow=/([A-Z][A-Z0-9\-\/\.]{3,30})\s*(?:\u2192|\u21d2|\u304b\u3089|\u3088\u308a)\s*([A-Z][A-Z0-9\-\/\.]{3,30})/g;
  for(const m of t.toUpperCase().matchAll(arrow)){

function monotaroPrimaryWindow(text="",pn=""){
  const src=String(text||"").normalize("NFKC");
  const rawPn=String(pn||"").normalize("NFKC").trim();
  const candidates=[rawPn,normalizePartNumber(rawPn)].filter(Boolean);
  let idx=-1;
  for(const c of candidates){
    idx=src.toUpperCase().indexOf(c.toUpperCase());
    if(idx>=0)break;
  }
  if(idx<0)return src.slice(0,5000);
  // The commercial facts for the selected SKU sit close to the exact MPN.
  // Keep recommendations/related-products outside the decision window.
  return src.slice(Math.max(0,idx-1800),Math.min(src.length,idx+2600));
}
function canonicalMonotaroProductUrl(url=""){
  try{
    const u=new URL(String(url));
    const m=u.pathname.match(/^\/p\/(\d{4})\/(\d{4})(?:\/print)?\/?$/);
    if(!m)return String(url);
    return `https://www.monotaro.com/p/${m[1]}/${m[2]}/`;
  }catch{return String(url);}
}
function monotaroPrintUrl(url=""){
  const canonical=canonicalMonotaroProductUrl(url);
  return /\/p\/\d{4}\/\d{4}\/$/.test(canonical)?`${canonical}print/`:null;
}
function classifyMonotaroCommerce({discontinued=false,salePrice=null,leadDays=null}={}){
  const hasSalePrice=salePrice!=null&&salePrice!==""&&Number.isFinite(Number(salePrice))&&Number(salePrice)>0;
  const hasLead=leadDays!=null&&leadDays!==""&&Number.isFinite(Number(leadDays))&&Number(leadDays)>=0;
  const active=!discontinued && (hasSalePrice||hasLead);
  return {
    active,
    stock_status:discontinued?"out_of_stock":hasLead?"in_stock":"unknown",
    price:active&&hasSalePrice?Number(salePrice):null
  };
}
function sliceLabelValue(src,label,nextLabels=[]){
  const hay=String(src||"");
  const idx=hay.indexOf(label);
  if(idx<0)return null;
  let v=hay.slice(idx+label.length).replace(/^\s*[:\uff1a]?\s*/,"");
  let cut=v.length;
  for(const n of nextLabels){
    const j=v.indexOf(n);
    if(j>=0&&j<cut)cut=j;
  }
  return v.slice(0,cut).trim();
}
