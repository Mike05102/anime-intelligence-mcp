  const e=best.purchase_evaluation||{},u=e.unknowns||[];
  if(e.hard_blockers?.length)return {action:"VERIFY_BEFORE_BUY",can_purchase:false,reason_codes:e.hard_blockers,best};
  if(e.identity?.known===false)return {action:"COMPARE_AND_CONFIRM_IDENTITY",can_purchase:false,reason_codes:["exact_identity_not_proven"],best};
  if(u.some(x=>["shipping","delivery_time","stock_status"].includes(x)))return {action:"VERIFY_FULFILLMENT",can_purchase:false,reason_codes:u.filter(x=>["shipping","delivery_time","stock_status"].includes(x)),best};

function universalCandidateRelevance(query="",o={}){

  const domainMetrics=new Map();for(const dname of UNIVERSAL_GENERIC_DOMAINS){const d=domains.get(dname)||{domain:dname,runs:0,candidates:0,unique:new Set(),gtins:0,canonical_new:0,merchant_evidence:0,merchant_trust:0,purchase_ready:0,decision_ready:0};domainMetrics.set(dname,{domain:dname,runs:d.runs,candidates:d.candidates,unique:d.unique.size,unique_yield:d.candidates?d.unique.size/d.candidates:null,gtin_yield:d.candidates?d.gtins/d.candidates:null,canonical_yield:d.candidates?d.canonical_new/d.candidates:null,merchant_evidence_yield:d.candidates?d.merchant_evidence/d.candidates:null,merchant_trust_yield:d.candidates?d.merchant_trust/d.candidates:null,purchase_ready_yield:d.candidates?d.purchase_ready/d.candidates:null,decision_ready_yield:d.candidates?d.decision_ready/d.candidates:null});}
  return {families:familyMetrics,domains:domainMetrics,recentFingerprints,totalRuns};
}
function adaptiveExplorationScore(candidate,history,now=Date.now()){
  const family=universalQueryFamily(candidate.domain,candidate.query),st=history?.families?.get(family),domain=history?.domains?.get(candidate.domain),runs=Number(st?.runs||0),uy=st?Number(st.unique_yield||0):1,gy=st?Number(st.gtin_yield||0):0,cy=st?Number(st.canonical_yield||0):0,my=st?Number(st.merchant_evidence_yield||0):0,mty=st?Number(st.merchant_trust_yield||0):0,py=st?Number(st.purchase_ready_yield||0):0,dy=st?Number(st.decision_ready_yield||0):0,ay=st?Number(st.decision_answerability_yield||0):0,ce=st?Number(st.canonical_per_provider_call||0):0,dup=st?Number(st.duplicate_ratio||0):0,age=st?Number(st.age_hours||0):Infinity;
  const sourceBase={demand:3.2,frontier:1.6,stale_refresh:1.0,exploration:1.7,long_tail:1.95,domain_gap:2.20}[candidate.source]||1.5;
  const total=Math.max(1,Number(history?.totalRuns||0)),ucb=Math.sqrt(Math.log(total+2)/(runs+1));
  const domainRuns=Number(domain?.runs||0),domainNovelty=1/Math.sqrt(domainRuns+1);
  let score=sourceBase+UNIVERSAL_DISCOVERY_POLICY.unique_yield_weight*uy+UNIVERSAL_DISCOVERY_POLICY.gtin_yield_weight*gy+UNIVERSAL_DISCOVERY_POLICY.canonical_yield_weight*cy+UNIVERSAL_DISCOVERY_POLICY.merchant_evidence_yield_weight*my+UNIVERSAL_DISCOVERY_POLICY.merchant_trust_yield_weight*mty+UNIVERSAL_DISCOVERY_POLICY.purchase_ready_yield_weight*py+UNIVERSAL_DISCOVERY_POLICY.decision_ready_yield_weight*dy+UNIVERSAL_DISCOVERY_POLICY.decision_answerability_weight*ay+UNIVERSAL_DISCOVERY_POLICY.canonical_efficiency_weight*Math.min(1,ce)+UNIVERSAL_DISCOVERY_POLICY.exploration_bonus_weight*ucb+0.45*domainNovelty-UNIVERSAL_DISCOVERY_POLICY.duplicate_penalty_weight*dup;
  if(runs>=UNIVERSAL_DISCOVERY_POLICY.max_query_family_runs_7d)score-=0.8;
  if(age<UNIVERSAL_DISCOVERY_POLICY.query_cooldown_hours)score-=1.0*(1-age/UNIVERSAL_DISCOVERY_POLICY.query_cooldown_hours);
  if(st&&uy<0.35&&age<UNIVERSAL_DISCOVERY_POLICY.low_yield_cooldown_hours)score-=1.2*(1-age/UNIVERSAL_DISCOVERY_POLICY.low_yield_cooldown_hours);
  if(st&&Number(st.candidates||0)>=50&&cy===0)score-=UNIVERSAL_DISCOVERY_POLICY.zero_canonical_penalty;if(st&&Number(st.candidates||0)>=50&&dy===0)score-=UNIVERSAL_DISCOVERY_POLICY.zero_decision_ready_penalty;
  const paidConversions=Number(history?.paidFamilies?.get(family)||0);if(paidConversions>0)score+=UNIVERSAL_DISCOVERY_POLICY.revenue_signal_weight*Math.min(2,Math.log2(1+paidConversions));
  if(candidate.pre_payment)score+=1.0;
  return {score:Math.round(score*1000)/1000,family,history_runs:runs,history_candidates:Number(st?.candidates||0),history_gtin_candidates:Number(st?.gtins||0),history_canonical_new_count:Number(st?.canonical_new||0),history_avg_candidates_per_run:runs?Math.round(Number(st?.candidates||0)/runs*1000)/1000:0,history_unique_yield:st?Math.round(uy*1000)/1000:null,history_duplicate_ratio:st?Math.round(dup*1000)/1000:null,history_gtin_yield:st?Math.round(gy*1000)/1000:null,history_canonical_yield:st?Math.round(cy*1000)/1000:null,history_merchant_evidence_yield:st?Math.round(my*1000)/1000:null,history_merchant_trust_yield:st?Math.round(mty*1000)/1000:null,history_purchase_ready_yield:st?Math.round(py*1000)/1000:null,history_decision_ready_yield:st?Math.round(dy*1000)/1000:null,history_decision_answerability_yield:st?Math.round(ay*1000)/1000:null,history_canonical_per_provider_call:st?Math.round(ce*1000)/1000:null,history_paid_conversions:paidConversions,last_run_age_hours:Number.isFinite(age)?Math.round(age*10)/10:null};
}

    const enrichedOffers=allOffers.map(o=>({...o}));
