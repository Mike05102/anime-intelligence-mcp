function identityOptimizedSearchQuery(query="",exactId="",domain="general_commerce"){
  const exact=String(exactId||"").trim();
  if(exact&&validGtin(digitsOnly(exact)))return digitsOnly(exact);
  let q=String(query||"")
    .replace(/[※◆■●★☆]/g," ")
    .replace(/\b(?:new|used|sealed|open box|free shipping|parallel import|imported|returns?|refund|warranty|guarantee|cheap|sale)\b/gi," ")
    .replace(/(?:中古|新品|未開封|並行輸入品?|送料無料|返品(?:返金)?不可|保証(?:なし|無し)?|動作未確認|お安くご提供|商品として)/g," ")
    .replace(/\s+/g," ").trim();
  // Remove long disclaimer tails that commonly follow punctuation in marketplace titles.
  q=q.split(/[|｜;；]/)[0].trim();
  // Keep strong model / part tokens and enough surrounding product words for Yahoo identity recall.
  const tokens=q.split(/\s+/).filter(Boolean);
  const modelTokens=tokens.filter(t=>/[A-Za-z]/.test(t)&&/\d/.test(t)&&t.length>=4).slice(0,2);
  const base=[];
  for(const t of tokens){
    if(base.length>=8)break;
    if(t.length<=1)continue;
    base.push(t);
  }
  for(const t of modelTokens)if(!base.includes(t))base.push(t);
  const out=base.join(" ").trim();
  return (out||q||String(query||"")).slice(0,100);
}
function identityQueryBreadthScore(query=""){
  const q=String(query||"").trim(),tokens=q.split(/\s+/).filter(Boolean);
  const longPenalty=Math.max(0,tokens.length-10)*0.08+Math.max(0,q.length-100)/250;
  const identifierOnly=/^(?:GTIN|JAN|EAN|UPC|ISBN)?\s*\d{8,14}$/i.test(q)?0.55:0;
  return Math.max(0,1-longPenalty-identifierOnly);
}
