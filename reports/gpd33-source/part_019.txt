function merchantFiniteNumber(v,field,{min=null,max=null,integer=false}={}){
  if(v==null||v==="")return null;
  const n=Number(v);
  if(!Number.isFinite(n))throw new Error(`invalid_${field}`);
  const out=integer?Math.floor(n):n;

  return out;
}

function isbn10Valid(v=""){
