async function sha256Hex(value=""){
  const data=new TextEncoder().encode(String(value));const digest=await crypto.subtle.digest("SHA-256",data);return [...new Uint8Array(digest)].map(b=>b.toString(16).padStart(2,"0")).join("");
}

// v3.7.92 — isolate known legacy self-tests without permanently blacklisting a generic Node/NL fingerprint.
// The fingerprint below was produced by our own pre-v3.7.92 test runner. It is treated as test traffic only
// for events at or before the final known self-test window. Future traffic with the same coarse fingerprint
// remains eligible to count as external unless it carries an explicit test marker.
const LEGACY_SELF_TEST_FINGERPRINT_CUTOFFS={
  "03ed7b5e9a95f7f08a07af23":"2026-09-24T08:28:51.132Z"

// v3.7.98 — exact known self-test signature. This is intentionally narrower than a permanent

  // Fall back to a broad recent read and filter in JavaScript so telemetry failure is never disguised as zero traffic.

  }catch(e){fallbackError=safeError(e);}
