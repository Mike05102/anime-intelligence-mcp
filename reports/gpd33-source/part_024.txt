function purchaseDeadlineDays(deadline){
  if(!deadline)return null;const t=Date.parse(deadline);if(!Number.isFinite(t))return null;
  return Math.max(0,Math.ceil((t-Date.now())/86400000));
}
function purchaseKnownOfferTotal(offer,quantity=1){
  const price=purchaseFinite(offer?.price);if(price==null)return null;
  const shipping=purchaseFinite(offer?.shipping_cost);
  // A landed total is only known when shipping is explicitly known. Missing shipping is not zero.
  if(shipping==null)return null;
  return price*Math.max(1,quantity)+shipping;
}
function assessPartMerchantOffer(x,intent){
  const o=x?.offer||{},m=x?.merchant||{},score=x?.agent_score||{};
  const itemSubtotal=purchaseFinite(o?.price)==null?null:purchaseFinite(o.price)*Math.max(1,intent.quantity);
  const total=purchaseKnownOfferTotal(o,intent.quantity);
  const currency=String(o.currency||"").toUpperCase()||null;
  const max=intent.requirements.max_total_cost,deadlineDays=purchaseDeadlineDays(intent.requirements.delivery_deadline);
  const exact=normalizePartNumber(o.part_number||o.mpn||"")===normalizePartNumber(intent.product.mpn||"");
  const evidenceBackedAuthorized=Boolean(o.authorized_distributor&&(m.verified||o.authenticity_evidence_url));
  const authorizedOk=!intent.requirements.authorized_only||evidenceBackedAuthorized;
  const authenticityOk=!intent.requirements.authenticity_required||Boolean(m.verified||o.authenticity_evidence_url);
  // A hard budget cannot pass on a currency mismatch or unknown landed total.
  const budgetOk=max==null||(currency===intent.requirements.currency&&total!=null&&total<=max);
  // A hard delivery deadline cannot pass when lead time is unknown.
  const deliveryOk=deadlineDays==null||(o.lead_time_days!=null&&Number.isFinite(Number(o.lead_time_days))&&Number(o.lead_time_days)<=deadlineDays);
  const stockOk=["in_stock","low_stock"].includes(String(o.stock_status||""));
  const moq=purchaseFinite(o?.global_fulfillment?.minimum_order_quantity);
  const quantityOk=moq==null||intent.quantity>=moq;
  const ships=(o.ships_to||[]).map(v=>String(v).toUpperCase());
  const destinationOk=!ships.length||ships.includes(intent.destination.country)||ships.includes("WORLDWIDE")||ships.includes("ALL");
  const eligible=exact&&authorizedOk&&authenticityOk&&budgetOk&&deliveryOk&&stockOk&&quantityOk&&destinationOk;
  const unknown=[...(o.shipping_cost==null?["shipping"]:[])];
  if(o?.global_fulfillment?.tax_included!==true)unknown.push("tax_or_vat");
  if(o?.global_fulfillment?.duty_included!==true&&intent.destination.country!==String(m.country_code||"").toUpperCase())unknown.push("customs_or_duty");
  if(intent.destination.country!==String(m.country_code||"").toUpperCase())unknown.push("brokerage_if_applicable");
  return {...x,purchase_evaluation:{eligible,exact_identity:exact,authorized_requirement_met:authorizedOk,authenticity_requirement_met:authenticityOk,budget_requirement_met:budgetOk,delivery_requirement_met:deliveryOk,stock_usable:stockOk,quantity_requirement_met:quantityOk,destination_requirement_met:destinationOk,item_subtotal:itemSubtotal,known_total:total,currency,unknown_cost_components:[...new Set(unknown)],overall_agent_score:Number(score.overall_agent_score||0)}};
}
function globalPartsOfferBlockers(x,intent){
  const e=x?.purchase_evaluation||{},o=x?.offer||{};
  const blockers=[];
  if(!e.exact_identity)blockers.push("identity_mismatch");
  if(!e.stock_usable)blockers.push("stock_unavailable");
  if(!e.authorized_requirement_met)blockers.push("authorized_status_not_verified");
  if(!e.authenticity_requirement_met)blockers.push("authenticity_evidence_missing");
  if(!e.delivery_requirement_met)blockers.push(o.lead_time_days==null?"lead_time_unknown":"delivery_deadline_not_met");
  if(!e.budget_requirement_met)blockers.push(e.known_total==null?"landed_total_unknown_or_shipping_unknown":"budget_exceeded_or_currency_mismatch");
  if(!o.product_url&&!o.checkout_url)blockers.push("purchase_url_missing");
  return [...new Set(blockers)];
}
function globalPartsDecision(partIntel,evaluated,intent){
  const eligible=evaluated.filter(x=>x.purchase_evaluation?.eligible);
  const best=eligible[0]||null;
  const exact=evaluated.filter(x=>x.purchase_evaluation?.exact_identity);
  const stocked=exact.filter(x=>x.purchase_evaluation?.stock_usable);
  const successors=partIntel?.relations?.successors||[];
  const substitutes=partIntel?.relations?.verified_substitutes||[];
  let action="NO_VERIFIED_PURCHASE_ROUTE",reason_codes=[];
  if(best){action="BUY_EXACT";reason_codes=["exact_mpn","merchant_offer_eligible","hard_constraints_met","evidence_based_ranking"]; }
  else if(stocked.length){
    const blockers=[...new Set(stocked.flatMap(x=>globalPartsOfferBlockers(x,intent)))];
    if(blockers.some(x=>["authorized_status_not_verified","authenticity_evidence_missing"].includes(x))){action="VERIFY_SELLER_EVIDENCE";reason_codes=blockers;}
    else {action="VERIFY_FULFILLMENT_OR_TOTAL";reason_codes=blockers.length?blockers:["hard_constraint_unknown"]; }
  }else if(exact.length){action="WAIT_FOR_STOCK";reason_codes=["exact_offer_exists_but_not_currently_buyable"]; }
  else if(intent.requirements.substitutes_allowed&&successors.length){action="REVIEW_VERIFIED_SUCCESSOR";reason_codes=["no_eligible_exact_offer","verified_successor_available"]; }
  else if(intent.requirements.substitutes_allowed&&substitutes.length){action="REVIEW_VERIFIED_SUBSTITUTE";reason_codes=["no_eligible_exact_offer","verified_substitute_available"]; }
  else if(!intent.requirements.substitutes_allowed){action="NO_ELIGIBLE_EXACT_OFFER";reason_codes=["substitution_not_allowed","no_exact_offer_meets_constraints"]; }
  else {action="NO_SAFE_SUBSTITUTE";reason_codes=["no_exact_offer","no_verified_successor_or_substitute"]; }
  const evidenceComplete=best?(!globalPartsOfferBlockers(best,intent).length):false;
  const confidence=best&&evidenceComplete?"high":stocked.length?"medium":exact.length?"medium_low":"low";
  return {action,can_purchase:!!best,requires_user_confirmation:true,reason_codes,decision_confidence:confidence,best,eligible_count:eligible.length,exact_offer_count:exact.length,compatibility_policy:"never_infer_substitute_from_name_or_dimensions"};
}

function rankPurchasePartOffers(items=[]){
  return [...items].sort((a,b)=>{
    const A=a.purchase_evaluation||{},B=b.purchase_evaluation||{};
    if(Boolean(A.eligible)!==Boolean(B.eligible))return B.eligible?1:-1;
    const sa=Number(A.overall_agent_score||0),sb=Number(B.overall_agent_score||0);if(sa!==sb)return sb-sa;
    if(A.currency&&A.currency===B.currency&&A.known_total!=null&&B.known_total!=null)return A.known_total-B.known_total;
    return 0;
  });
}
