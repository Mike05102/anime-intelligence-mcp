function rsStockMap(label=""){
  const x=String(label||"").trim();

function parseRsOmronOffers(source,page){
  const text=String(page.text||stripHtml(page.html||"")).normalize("NFKC");
  const statuses=["\u5728\u5eab\u3042\u308a","\u5728\u5eab\u9650\u308a","\u304a\u53d6\u308a\u5bc4\u305b\u54c1","\u53d7\u6ce8\u8abf\u9054","\u5165\u8377\u5f85\u3061\uff08\u65b0\u5546\u54c1\uff09","\u4e00\u6642\u7684\u306b\u5728\u5eab\u5207\u308c","\u5165\u8377\u5f85\u3061","\u53d6\u6271\u505c\u6b62\u4e2d"];

function decodeEmbeddedUnicode(v=""){
  return String(v||"")
    .replace(/\\u\{([0-9a-fA-F]+)\}/g,(_,h)=>{try{return String.fromCodePoint(parseInt(h,16));}catch{return _;}})
    .replace(/\\u([0-9a-fA-F]{4})/g,(_,h)=>String.fromCharCode(parseInt(h,16)))
    .replace(/\\x([0-9a-fA-F]{2})/g,(_,h)=>String.fromCharCode(parseInt(h,16)));
}
function monotaroPrimaryWindow(text="",pn=""){
  const src=String(text||"").normalize("NFKC");
  const rawPn=String(pn||"").normalize("NFKC").trim();
  const candidates=[rawPn,normalizePartNumber(rawPn)].filter(Boolean);
  let idx=-1;
  for(const c of candidates){
    idx=src.toUpperCase().indexOf(c.toUpperCase());
    if(idx>=0)break;
  }
  if(idx<0)return src.slice(0,5000);
  // The commercial facts for the selected SKU sit close to the exact MPN.
  // Keep recommendations/related-products outside the decision window.
  return src.slice(Math.max(0,idx-1800),Math.min(src.length,idx+2600));
}

function classifyMonotaroCommerce({discontinued=false,salePrice=null,leadDays=null}={}){
  const hasSalePrice=salePrice!=null&&salePrice!==""&&Number.isFinite(Number(salePrice))&&Number(salePrice)>0;
  const hasLead=leadDays!=null&&leadDays!==""&&Number.isFinite(Number(leadDays))&&Number(leadDays)>=0;
  const active=!discontinued && (hasSalePrice||hasLead);

function sliceLabelValue(src,label,nextLabels=[]){
  const hay=String(src||"");
  const idx=hay.indexOf(label);
  if(idx<0)return null;
  let v=hay.slice(idx+label.length).replace(/^\s*[:\uff1a]?\s*/,"");
  let cut=v.length;
  for(const n of nextLabels){
    const j=v.indexOf(n);
    if(j>=0&&j<cut)cut=j;
  }
  return v.slice(0,cut).trim();
}

function parseMonotaroEvidenceText(text="",pageUrl=""){

function parseMonotaroOffers(source,page){
  const html=String(page.html||"");
  const text=String(page.text||stripHtml(html)).normalize("NFKC");
  const raw=html.normalize("NFKC");
  const decodedRaw=decodeEmbeddedUnicode(raw).normalize("NFKC");
  const pageUrl=String(page.official_url||page.url||"");
  if(/\/s\//.test(pageUrl))return [];
  if(!/\/p\//.test(pageUrl))return [];

  // Print pages are the canonical commerce evidence path for MonotaRO.
  // Use one pure parser for both production and self-test to prevent runtime drift.
  const combined=`${text}\n${stripHtml(decodedRaw)}`.normalize("NFKC");
  if(/\/print\/?(?:[?#].*)?$/.test(pageUrl)){
    const parsed=parseMonotaroEvidenceText(combined,pageUrl);
    return parsed?[parsed]:[];
  }

  // Normal product pages are identity/discovery fallback only. Keep fail-closed.
  const title=html?String(titleOf(html)||"").normalize("NFKC"):(text.match(/^Title:\s*(.+)$/mi)?.[1]||"");
  const explicitText=combined.match(/(?:\u30e1\u30fc\u30ab\u30fc)?\u54c1\u756a\s*[:\uff1a]?\s*([A-Z0-9][A-Z0-9._/+()\-]{1,100}(?:\s+\d+M)?)/i);

  const displayPn=(explicitText?.[1]||titleLead?.[1]||"").trim().replace(/\s+/g," ");
  const normalizedPn=normalizePartNumber(displayPn);
  if(!normalizedPn || !/[A-Z]/.test(normalizedPn) || !/[0-9]/.test(normalizedPn))return [];
  const pathSku=pageUrl.match(/\/p\/(\d{4})\/(\d{4})\//)?.slice(1).join("")||null;
  return [{
    part_number:displayPn||normalizedPn,
    price:null,
    reference_price:null,
    availability_label:null,
    stock_status:"unknown",
    active:false,
    tax_included:false,

    merchant_sku:pathSku,
    identity_evidence:explicitText?"visible_part_number":"document_title_exact_token",

  }];
}
