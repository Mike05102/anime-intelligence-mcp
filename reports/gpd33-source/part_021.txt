function universalIdentityRoleScore(x={}){
  const n=Math.max(0,Number(x.history_candidates||0)),runs=Math.max(0,Number(x.history_runs||0));
  const gtins=Math.max(0,Number(x.history_gtin_candidates||0)),canonical=Math.max(0,Number(x.history_canonical_new_count||0));
  const avg=Math.max(0,Number(x.history_avg_candidates_per_run||(runs?n/runs:0)));
  const uy=Number(x.history_unique_yield??0.5),ce=Number(x.history_canonical_per_provider_call||0);
  // Beta-style shrinkage prevents a 1/1 exact listing from outranking a broad query that yields many canonical identities.
  const gy=(gtins+1.5)/(n+12.5),cy=(canonical+0.75)/(n+12.5);
  const volume=Math.min(1,Math.log1p(avg)/Math.log(31));
  const expectedCanonical=Math.min(1,(cy*Math.max(1,avg))/6);
  const expectedGtin=Math.min(1,(gy*Math.max(1,avg))/10);
  const breadth=identityQueryBreadthScore(x.query||"");
  return 0.28*expectedGtin+0.34*expectedCanonical+0.18*volume+0.08*uy+0.07*Math.min(1,ce)+0.05*breadth+(runs===0?0.08:0);
}
function universalFulfillmentRoleScore(x={}){
  const dy=Number(x.history_decision_ready_yield||0),mty=Number(x.history_merchant_trust_yield||0),py=Number(x.history_purchase_ready_yield||0),uy=Number(x.history_unique_yield??0.5),runs=Number(x.history_runs||0);
  return UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_decision_weight*dy+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_trust_weight*mty+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_purchase_weight*py+UNIVERSAL_DISCOVERY_POLICY.fulfillment_role_unique_weight*uy+(runs===0?0.05:0);
}
function universalCrossProviderRoleScore(x={}){return 0.65*universalIdentityRoleScore(x)+0.35*universalFulfillmentRoleScore(x);}
