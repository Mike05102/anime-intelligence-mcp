    created_at:now,updated_at:now
  };
  const saved=await sb(env,"/merchants",{method:"POST",headers:{Prefer:"return=representation"},body:JSON.stringify(row)});
  const merchant=Array.isArray(saved)&&saved.length?saved[0]:row;
  await sbOptional(env,"/merchant_agent_profiles?on_conflict=merchant_id",{method:"POST",headers:{Prefer:"resolution=merge-duplicates,return=minimal"},body:JSON.stringify({
