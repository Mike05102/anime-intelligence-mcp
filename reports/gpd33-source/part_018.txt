  if(o.datasheet_url)trust+=15;

async function enforceMerchantRegistrationRateLimit(env,request){
  const ipHash=await merchantRegistrationIpHash(request);
  if(!ipHash)return {ip_hash:null,recent_count:null};
  const cutoff=new Date(Date.now()-24*60*60*1000).toISOString();
  const path=`/merchants?select=id&metadata-%3E%3Eregistration_ip_hash=eq.${encodeURIComponent(ipHash)}&created_at=gte.${encodeURIComponent(cutoff)}&limit=4`;
  const rows=await sbOptional(env,path);
  const count=Array.isArray(rows)?rows.length:0;
  if(count>=3){
    const e=new Error("merchant_registration_rate_limited");
    e.status=429;
    throw e;
  }
  return {ip_hash:ipHash,recent_count:count};
}
